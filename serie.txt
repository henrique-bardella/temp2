Histórico Indicadores HTML =
VAR _W = 1400
VAR _H = 500

VAR _PadLeft = 55
VAR _PadRight = 30
VAR _PadTop = 45
VAR _PadBottom = 70

VAR _ChartW = _W - _PadLeft - _PadRight
VAR _ChartH = _H - _PadTop - _PadBottom


/* ============================================================
   FIELD PARAMETER
   IMPORTANTE:
   SUMMARIZE inclui TODA a chave composta do parâmetro.
   Depois SELECTCOLUMNS cria colunas virtuais simples.
   ============================================================ */

VAR _IndicadoresSelecionados =
    SELECTCOLUMNS (
        SUMMARIZE (
            'Parâmetro',
            'Parâmetro'[Parâmetro],
            'Parâmetro'[Parâmetro Fields],
            'Parâmetro'[Parâmetro Order]
        ),
        "@Indicador", 'Parâmetro'[Parâmetro],
        "@Ordem", 'Parâmetro'[Parâmetro Order]
    )


/* ============================================================
   PERÍODOS VISÍVEIS
   Mantém os demais filtros do relatório.
   Remove somente o contexto individual de ANO/MESANO
   para reconstruir a série histórica visível.
   ============================================================ */

VAR _PeriodosBase =
    CALCULATETABLE (
        SUMMARIZE (
            'BASE',
            'BASE'[ANO],
            'BASE'[MESANO]
        ),
        ALLSELECTED (
            'BASE'[ANO],
            'BASE'[MESANO]
        )
    )


/* ============================================================
   POSIÇÃO CRONOLÓGICA
   MESANO deve ser uma data ou coluna cronologicamente ordenável
   ============================================================ */

VAR _Periodos =
    ADDCOLUMNS (
        _PeriodosBase,
        "@Pos",
            RANKX (
                _PeriodosBase,
                'BASE'[MESANO],
                ,
                ASC,
                DENSE
            )
    )

VAR _QtdPeriodos =
    COUNTROWS ( _Periodos )

VAR _DivX =
    MAX (
        1,
        _QtdPeriodos - 1
    )


/* ============================================================
   CRIA UMA TABELA LONGA

   Indicador | Ano | MesAno | Pos | Valor

   Isso permite trabalhar com todas as séries dinamicamente.
   ============================================================ */

VAR _DadosLongos =
    GENERATE (
        _IndicadoresSelecionados,

        VAR _NomeIndicador =
            [@Indicador]

        VAR _OrdemIndicador =
            [@Ordem]

        VAR _CorIndicador =
            SWITCH (
                _NomeIndicador,

                "Resultado - ROB Crédito",
                    "#005CA9",

                "Resultado - ROB Não crédito",
                    "#172B4D",

                "Resultado - TPV",
                    "#45A6CF",

                "Resultado - Fat. Líquido",
                    "#83C5E3",

                "Consistência - Encanta/NPS",
                    "#2D6A4F",

                "Consistência - BSC",
                    "#5C8D3E",

                "PDD",
                    "#777777",

                "#555555"
            )

        /*
           Alterna a posição dos labels para reduzir
           sobreposição entre as séries.
        */

        VAR _OffsetLabel =
            SWITCH (
                MOD ( _OrdemIndicador, 4 ),
                0, -11,
                1, 20,
                2, -23,
                3, 32,
                -11
            )

        RETURN

            SELECTCOLUMNS (
                ADDCOLUMNS (
                    _Periodos,

                    "@ValorTemp",

                        SWITCH (
                            _NomeIndicador,

                            "Resultado - ROB Crédito",
                                CALCULATE (
                                    'BASE'[% Atingimento ROB Crédito]
                                ) * 100,

                            "Resultado - ROB Não crédito",
                                CALCULATE (
                                    'BASE'[% Atingimento ROB Não Crédito]
                                ) * 100,

                            "Resultado - TPV",
                                CALCULATE (
                                    'BASE'[% Atingimento TPV]
                                ) * 100,

                            "Resultado - Fat. Líquido",
                                CALCULATE (
                                    'BASE'[% Atingimento Faturamento Liquido]
                                ) * 100,

                            "Consistência - Encanta/NPS",
                                CALCULATE (
                                    'BASE'[% Atingimento Encanta BRA]
                                ) * 100,

                            "Consistência - BSC",
                                CALCULATE (
                                    'BASE'[% Atingimento BSC]
                                ) * 100,

                            "PDD",
                                CALCULATE (
                                    'BASE'[% Atingimento PDD]
                                ) * 100
                        )
                ),

                "@IndicadorL",
                    _NomeIndicador,

                "@OrdemL",
                    _OrdemIndicador,

                "@CorL",
                    _CorIndicador,

                "@OffsetL",
                    _OffsetLabel,

                "@PosL",
                    [@Pos],

                "@AnoL",
                    'BASE'[ANO],

                "@MesAnoL",
                    'BASE'[MESANO],

                "@ValorL",
                    [@ValorTemp]
            )
    )


/* ============================================================
   ESCALA Y
   ============================================================ */

VAR _MinData =
    MINX (
        FILTER (
            _DadosLongos,
            NOT ISBLANK ( [@ValorL] )
        ),
        [@ValorL]
    )

VAR _MaxData =
    MAXX (
        FILTER (
            _DadosLongos,
            NOT ISBLANK ( [@ValorL] )
        ),
        [@ValorL]
    )


/*
   Força 100% a permanecer dentro da escala.
*/

VAR _MinBase =
    MIN (
        _MinData,
        100
    )

VAR _MaxBase =
    MAX (
        _MaxData,
        100
    )

VAR _Amplitude =
    MAX (
        10,
        _MaxBase - _MinBase
    )

VAR _YMin =
    MAX (
        0,
        _MinBase - _Amplitude * 0.15
    )

VAR _YMax =
    _MaxBase + _Amplitude * 0.15

VAR _YRange =
    MAX (
        1,
        _YMax - _YMin
    )


/* ============================================================
   GRID HORIZONTAL
   ============================================================ */

VAR _Grid =
    CONCATENATEX (
        GENERATESERIES (
            0,
            4,
            1
        ),

        VAR _GridIndex =
            [Value]

        VAR _GridValue =
            _YMin
                + DIVIDE (
                    _YRange,
                    4
                ) * _GridIndex

        VAR _Y =
            _PadTop
                + _ChartH
                - DIVIDE (
                    _GridValue - _YMin,
                    _YRange
                ) * _ChartH

        RETURN

            "
            <line
                x1='" & FORMAT ( _PadLeft, "0", "en-US" ) & "'
                x2='" & FORMAT ( _W - _PadRight, "0", "en-US" ) & "'
                y1='" & FORMAT ( _Y, "0.0", "en-US" ) & "'
                y2='" & FORMAT ( _Y, "0.0", "en-US" ) & "'
                stroke='#E8EAED'
                stroke-width='1'
            />

            <text
                x='" & FORMAT ( _PadLeft - 10, "0", "en-US" ) & "'
                y='" & FORMAT ( _Y + 4, "0.0", "en-US" ) & "'
                text-anchor='end'
                font-family='Segoe UI'
                font-size='10'
                fill='#8A8A8A'
            >
                " & FORMAT ( _GridValue, "0" ) & "%
            </text>
            ",

        ""
    )


/* ============================================================
   LINHA DE REFERÊNCIA = 100%
   ============================================================ */

VAR _Y100 =
    _PadTop
        + _ChartH
        - DIVIDE (
            100 - _YMin,
            _YRange
        ) * _ChartH

VAR _Linha100 =
    "
    <line
        x1='" & FORMAT ( _PadLeft, "0", "en-US" ) & "'
        x2='" & FORMAT ( _W - _PadRight, "0", "en-US" ) & "'
        y1='" & FORMAT ( _Y100, "0.0", "en-US" ) & "'
        y2='" & FORMAT ( _Y100, "0.0", "en-US" ) & "'
        stroke='#A94442'
        stroke-width='1.4'
        stroke-dasharray='6 6'
        opacity='0.85'
    />

    <text
        x='" & FORMAT ( _PadLeft - 10, "0", "en-US" ) & "'
        y='" & FORMAT ( _Y100 - 6, "0.0", "en-US" ) & "'
        text-anchor='end'
        font-family='Segoe UI'
        font-size='9'
        font-weight='700'
        fill='#A94442'
    >
        META
    </text>
    "


/* ============================================================
   EIXO X - MESES
   ============================================================ */

VAR _MesesSVG =
    CONCATENATEX (
        _Periodos,

        VAR _Pos =
            [@Pos]

        VAR _X =
            _PadLeft
                + DIVIDE (
                    _Pos - 1,
                    _DivX
                ) * _ChartW

        VAR _LabelMes =
            FORMAT (
                'BASE'[MESANO],
                "mmm/yyyy"
            )

        RETURN

            "
            <text
                x='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                y='" & FORMAT ( _H - 38, "0", "en-US" ) & "'
                text-anchor='middle'
                font-family='Segoe UI'
                font-size='10'
                fill='#747474'
            >
                " & _LabelMes & "
            </text>
            ",

        "",
        [@Pos],
        ASC
    )


/* ============================================================
   ANOS
   ============================================================ */

VAR _Anos =
    DISTINCT (
        SELECTCOLUMNS (
            _Periodos,
            "@Ano",
                'BASE'[ANO]
        )
    )


VAR _AnosSVG =
    CONCATENATEX (
        _Anos,

        VAR _AnoAtual =
            [@Ano]

        VAR _Inicio =
            MINX (
                FILTER (
                    _Periodos,
                    'BASE'[ANO] = _AnoAtual
                ),
                [@Pos]
            )

        VAR _Fim =
            MAXX (
                FILTER (
                    _Periodos,
                    'BASE'[ANO] = _AnoAtual
                ),
                [@Pos]
            )

        VAR _XInicio =
            _PadLeft
                + DIVIDE (
                    _Inicio - 1,
                    _DivX
                ) * _ChartW

        VAR _XFim =
            _PadLeft
                + DIVIDE (
                    _Fim - 1,
                    _DivX
                ) * _ChartW

        VAR _XCentro =
            DIVIDE (
                _XInicio + _XFim,
                2
            )

        RETURN

            "
            <text
                x='" & FORMAT ( _XCentro, "0.0", "en-US" ) & "'
                y='" & FORMAT ( _H - 10, "0", "en-US" ) & "'
                text-anchor='middle'
                font-family='Segoe UI'
                font-size='11'
                font-weight='600'
                fill='#474747'
            >
                " & FORMAT ( _AnoAtual, "0" ) & "
            </text>
            ",

        ""
    )


/* ============================================================
   LINHAS VERTICAIS DE SEPARAÇÃO DE ANO
   ============================================================ */

VAR _SeparadoresAno =
    CONCATENATEX (
        FILTER (
            _Anos,
            [@Ano]
                <> MINX (
                    _Anos,
                    [@Ano]
                )
        ),

        VAR _AnoAtual =
            [@Ano]

        VAR _PrimeiraPos =
            MINX (
                FILTER (
                    _Periodos,
                    'BASE'[ANO] = _AnoAtual
                ),
                [@Pos]
            )

        VAR _XAtual =
            _PadLeft
                + DIVIDE (
                    _PrimeiraPos - 1,
                    _DivX
                ) * _ChartW

        VAR _XAnterior =
            _PadLeft
                + DIVIDE (
                    _PrimeiraPos - 2,
                    _DivX
                ) * _ChartW

        VAR _XSeparador =
            DIVIDE (
                _XAtual + _XAnterior,
                2
            )

        RETURN

            "
            <line
                x1='" & FORMAT ( _XSeparador, "0.0", "en-US" ) & "'
                x2='" & FORMAT ( _XSeparador, "0.0", "en-US" ) & "'
                y1='" & FORMAT ( _PadTop, "0", "en-US" ) & "'
                y2='" & FORMAT ( _H - 27, "0", "en-US" ) & "'
                stroke='#D8DADD'
                stroke-width='1'
                stroke-dasharray='3 4'
            />
            ",

        ""
    )


/* ============================================================
   LEGENDA
   ============================================================ */

VAR _Legenda =
    CONCATENATEX (
        _IndicadoresSelecionados,

        VAR _Nome =
            [@Indicador]

        VAR _Ordem =
            [@Ordem]

        VAR _Cor =
            SWITCH (
                _Nome,

                "Resultado - ROB Crédito",
                    "#005CA9",

                "Resultado - ROB Não crédito",
                    "#172B4D",

                "Resultado - TPV",
                    "#45A6CF",

                "Resultado - Fat. Líquido",
                    "#83C5E3",

                "Consistência - Encanta/NPS",
                    "#2D6A4F",

                "Consistência - BSC",
                    "#5C8D3E",

                "PDD",
                    "#777777",

                "#555555"
            )

        RETURN

            "
            <div class='legend-item'>

                <span
                    class='legend-dot'
                    style='background:" & _Cor & ";'
                ></span>

                <span>
                    " & _Nome & "
                </span>

            </div>
            ",

        "",
        [@Ordem],
        ASC
    )


/* ============================================================
   DESENHA TODAS AS SÉRIES
   ============================================================ */

VAR _SeriesSVG =
    CONCATENATEX (
        _IndicadoresSelecionados,

        VAR _Nome =
            [@Indicador]

        VAR _Ordem =
            [@Ordem]

        VAR _Cor =
            SWITCH (
                _Nome,

                "Resultado - ROB Crédito",
                    "#005CA9",

                "Resultado - ROB Não crédito",
                    "#172B4D",

                "Resultado - TPV",
                    "#45A6CF",

                "Resultado - Fat. Líquido",
                    "#83C5E3",

                "Consistência - Encanta/NPS",
                    "#2D6A4F",

                "Consistência - BSC",
                    "#5C8D3E",

                "PDD",
                    "#777777",

                "#555555"
            )

        VAR _OffsetLabel =
            SWITCH (
                MOD ( _Ordem, 4 ),
                0, -11,
                1, 20,
                2, -23,
                3, 32,
                -11
            )

        VAR _DadosSerie =
            FILTER (
                _DadosLongos,
                [@IndicadorL] = _Nome
                    && NOT ISBLANK ( [@ValorL] )
            )


        /* ----------------------------------------------------
           PONTOS PARA POLYLINE
           ---------------------------------------------------- */

        VAR _Points =
            CONCATENATEX (
                _DadosSerie,

                VAR _Pos =
                    [@PosL]

                VAR _Valor =
                    [@ValorL]

                VAR _X =
                    _PadLeft
                        + DIVIDE (
                            _Pos - 1,
                            _DivX
                        ) * _ChartW

                VAR _Y =
                    _PadTop
                        + _ChartH
                        - DIVIDE (
                            _Valor - _YMin,
                            _YRange
                        ) * _ChartH

                RETURN

                    FORMAT (
                        _X,
                        "0.0",
                        "en-US"
                    )
                    & ","
                    &
                    FORMAT (
                        _Y,
                        "0.0",
                        "en-US"
                    ),

                " ",
                [@PosL],
                ASC
            )


        /* ----------------------------------------------------
           CÍRCULOS
           ---------------------------------------------------- */

        VAR _Circulos =
            CONCATENATEX (
                _DadosSerie,

                VAR _Pos =
                    [@PosL]

                VAR _Valor =
                    [@ValorL]

                VAR _X =
                    _PadLeft
                        + DIVIDE (
                            _Pos - 1,
                            _DivX
                        ) * _ChartW

                VAR _Y =
                    _PadTop
                        + _ChartH
                        - DIVIDE (
                            _Valor - _YMin,
                            _YRange
                        ) * _ChartH

                RETURN

                    "
                    <circle
                        cx='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                        cy='" & FORMAT ( _Y, "0.0", "en-US" ) & "'
                        r='4'
                        fill='" & _Cor & "'
                        stroke='#FFFFFF'
                        stroke-width='1.5'
                    />
                    ",

                "",
                [@PosL],
                ASC
            )


        /* ----------------------------------------------------
           LABELS DE VALOR
           ---------------------------------------------------- */

        VAR _Labels =
            CONCATENATEX (
                _DadosSerie,

                VAR _Pos =
                    [@PosL]

                VAR _Valor =
                    [@ValorL]

                VAR _X =
                    _PadLeft
                        + DIVIDE (
                            _Pos - 1,
                            _DivX
                        ) * _ChartW

                VAR _Y =
                    _PadTop
                        + _ChartH
                        - DIVIDE (
                            _Valor - _YMin,
                            _YRange
                        ) * _ChartH

                RETURN

                    "
                    <text
                        x='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                        y='" & FORMAT ( _Y + _OffsetLabel, "0.0", "en-US" ) & "'
                        text-anchor='middle'
                        font-family='Segoe UI'
                        font-size='9.5'
                        font-weight='600'
                        fill='" & _Cor & "'
                    >
                        " & FORMAT ( _Valor, "0.#" ) & "%
                    </text>
                    ",

                "",
                [@PosL],
                ASC
            )

        RETURN

            "
            <polyline
                points='" & _Points & "'
                fill='none'
                stroke='" & _Cor & "'
                stroke-width='2.3'
                stroke-linecap='round'
                stroke-linejoin='round'
                opacity='0.95'
            />
            "
            &
            _Circulos
            &
            _Labels,

        "",
        [@Ordem],
        ASC
    )


/* ============================================================
   QUANTIDADE DE INDICADORES SELECIONADOS
   ============================================================ */

VAR _QtdIndicadores =
    COUNTROWS (
        _IndicadoresSelecionados
    )


/* ============================================================
   CASO NÃO HAJA DADOS
   ============================================================ */

VAR _TemDados =
    COUNTROWS (
        FILTER (
            _DadosLongos,
            NOT ISBLANK ( [@ValorL] )
        )
    ) > 0


/* ============================================================
   HTML FINAL
   ============================================================ */

RETURN

IF (
    NOT _TemDados,

    "
    <div style='
        width:100%;
        height:100%;
        display:flex;
        align-items:center;
        justify-content:center;
        font-family:Segoe UI;
        color:#777;
        font-size:13px;
    '>
        Não há dados para os filtros selecionados.
    </div>
    ",

    "
    <div class='historico-card'>

        <style>

            .historico-card{
                width:100%;
                height:100%;
                background:#FFFFFF;
                font-family:'Segoe UI',Arial,sans-serif;
                box-sizing:border-box;
                overflow:hidden;
                padding:14px 16px 8px 16px;
            }

            .historico-title{
                font-size:18px;
                line-height:1.1;
                font-weight:700;
                color:#202020;
            }

            .historico-subtitle{
                margin-top:4px;
                font-size:11px;
                line-height:1.2;
                font-weight:400;
                color:#777777;
            }

            .legend{
                display:flex;
                flex-wrap:wrap;
                align-items:center;
                gap:8px 15px;
                margin-top:12px;
                margin-bottom:2px;
            }

            .legend-item{
                display:flex;
                align-items:center;
                gap:5px;
                font-size:10px;
                line-height:1;
                color:#555555;
                white-space:nowrap;
            }

            .legend-dot{
                width:8px;
                height:8px;
                min-width:8px;
                border-radius:50%;
                display:inline-block;
            }

            .chart{
                width:100%;
                margin-top:5px;
            }

            svg{
                display:block;
                width:100%;
                height:auto;
                overflow:visible;
            }

        </style>


        <div class='historico-title'>
            Percentuais de Atingimentos
        </div>

        <div class='historico-subtitle'>
            Atingimentos dos indicadores por mês e ano
        </div>


        <div class='legend'>
            "
            &
            _Legenda
            &
            "
        </div>


        <div class='chart'>

            <svg
                viewBox='0 0 "
                    & FORMAT ( _W, "0", "en-US" )
                    & " "
                    & FORMAT ( _H, "0", "en-US" )
                    & "'
                preserveAspectRatio='xMidYMid meet'
            >

                "
                &
                _Grid
                &
                _Linha100
                &
                _SeparadoresAno
                &
                _SeriesSVG
                &
                _MesesSVG
                &
                _AnosSVG
                &
                "

            </svg>

        </div>

    </div>
    "
)
