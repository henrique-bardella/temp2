Histórico Indicadores HTML =
VAR _W = 1600
VAR _SvgH = 470

VAR _PadLeft = 58
VAR _PadRight = 35
VAR _PadTop = 68
VAR _PadBottom = 72

VAR _ChartW = _W - _PadLeft - _PadRight
VAR _ChartH = _SvgH - _PadTop - _PadBottom

/* ============================================================
   CORES
   ============================================================ */

VAR _CorROB_Credito       = "#1769B0"
VAR _CorROB_NaoCredito    = "#162A86"
VAR _CorTPV               = "#84CBEF"
VAR _CorFatLiquido        = "#A9B7E8"

VAR _CorBSC               = "#49A942"
VAR _CorEncanta           = "#245B2B"

VAR _CorPDD               = "#747474"

VAR _CorTexto             = "#252525"
VAR _CorSub               = "#676767"
VAR _CorMuted             = "#818181"

VAR _CorGrid              = "#E8E8E8"
VAR _CorMeta              = "#A84442"

VAR _CorBest              = "#3F7D48"
VAR _CorWorst             = "#A8524B"

VAR _CorFuncao            = "#C96A3A"


/* ============================================================
   FIELD PARAMETER
   ============================================================ */

VAR _IndicadoresSelecionados =
    SELECTCOLUMNS (
        SUMMARIZE (
            'Parâmetro',
            'Parâmetro'[Parâmetro],
            'Parâmetro'[Parâmetro Campos],
            'Parâmetro'[Parâmetro Pedido]
        ),
        "@Indicador",
            'Parâmetro'[Parâmetro],

        "@Ordem",
            'Parâmetro'[Parâmetro Pedido]
    )

VAR _QtdIndicadores =
    COUNTROWS ( _IndicadoresSelecionados )

VAR _IndicadorUnico =
    IF (
        _QtdIndicadores = 1,
        MAXX (
            _IndicadoresSelecionados,
            [@Indicador]
        )
    )


/* ============================================================
   PERÍODOS
   ============================================================ */

VAR _PeriodosBase =
    CALCULATETABLE (
        SUMMARIZE (
            'BASE',
            'BASE'[ANO],
            'BASE'[MESANO]
        ),
        ALLSELECTED (
            'BASE'[ANO],
            'BASE'[MESANO]
        )
    )


/* ============================================================
   MÊS NUMÉRICO
   ============================================================ */

VAR _PeriodosMes =
    ADDCOLUMNS (
        _PeriodosBase,

        "@MesNum",

            VAR _M =
                LOWER (
                    TRIM (
                        'BASE'[MESANO] & ""
                    )
                )

            RETURN
                SWITCH (
                    TRUE (),
                    LEFT ( _M, 3 ) = "jan", 1,
                    LEFT ( _M, 3 ) = "fev", 2,
                    LEFT ( _M, 3 ) = "mar", 3,
                    LEFT ( _M, 3 ) = "abr", 4,
                    LEFT ( _M, 3 ) = "mai", 5,
                    LEFT ( _M, 3 ) = "jun", 6,
                    LEFT ( _M, 3 ) = "jul", 7,
                    LEFT ( _M, 3 ) = "ago", 8,
                    LEFT ( _M, 3 ) = "set", 9,
                    LEFT ( _M, 3 ) = "out", 10,
                    LEFT ( _M, 3 ) = "nov", 11,
                    LEFT ( _M, 3 ) = "dez", 12,
                    BLANK ()
                )
    )


VAR _PeriodosSort =
    ADDCOLUMNS (
        _PeriodosMes,

        "@Sort",
            VALUE ( 'BASE'[ANO] ) * 100
                + [@MesNum],

        "@DataRef",
            DATE (
                VALUE ( 'BASE'[ANO] ),
                [@MesNum],
                1
            )
    )


VAR _Periodos =
    ADDCOLUMNS (
        _PeriodosSort,

        "@Pos",

            VAR _Atual =
                [@Sort]

            RETURN
                RANKX (
                    _PeriodosSort,
                    [@Sort],
                    _Atual,
                    ASC,
                    DENSE
                )
    )


VAR _QtdPeriodos =
    COUNTROWS ( _Periodos )

VAR _DivX =
    MAX (
        1,
        _QtdPeriodos - 1
    )


/* ============================================================
   DATA DE INÍCIO DA FUNÇÃO
   ============================================================ */

VAR _DataFuncao =
    SELECTEDVALUE (
        'dimGerentes2'[data_funcao]
    )


VAR _DataFuncaoMes =
    IF (
        NOT ISBLANK ( _DataFuncao ),

        DATE (
            YEAR ( _DataFuncao ),
            MONTH ( _DataFuncao ),
            1
        )
    )


VAR _MinDataSerie =
    MINX (
        _Periodos,
        [@DataRef]
    )


VAR _MaxDataSerie =
    MAXX (
        _Periodos,
        [@DataRef]
    )


VAR _DataFuncaoDentro =
    NOT ISBLANK ( _DataFuncaoMes )
        && _DataFuncaoMes >= _MinDataSerie
        && _DataFuncaoMes <= _MaxDataSerie


VAR _PosFuncao =
    MINX (
        TOPN (
            1,
            FILTER (
                _Periodos,
                [@DataRef] >= _DataFuncaoMes
            ),
            [@DataRef],
            ASC
        ),
        [@Pos]
    )


/* ============================================================
   TABELA LONGA
   ============================================================ */

VAR _DadosLongos =
    GENERATE (
        _IndicadoresSelecionados,

        VAR _Nome =
            [@Indicador]

        VAR _Ordem =
            [@Ordem]

        VAR _Cor =
            SWITCH (
                _Nome,

                "Resultado - ROB Crédito",
                    _CorROB_Credito,

                "Resultado - ROB Não crédito",
                    _CorROB_NaoCredito,

                "Resultado - TPV",
                    _CorTPV,

                "Resultado - Fat. Líquido",
                    _CorFatLiquido,

                "Consistência - BSC",
                    _CorBSC,

                "Consistência - Encanta/NPS",
                    _CorEncanta,

                "PDD",
                    _CorPDD,

                "#777777"
            )

        RETURN

            SELECTCOLUMNS (
                ADDCOLUMNS (
                    _Periodos,

                    "@ValorTemp",

                        SWITCH (
                            _Nome,

                            "Resultado - ROB Crédito",
                                CALCULATE (
                                    'BASE'[% Atingimento ROB Crédito]
                                ) * 100,

                            "Resultado - ROB Não crédito",
                                CALCULATE (
                                    'BASE'[% Atingimento ROB Não Crédito]
                                ) * 100,

                            "Resultado - TPV",
                                CALCULATE (
                                    'BASE'[% Atingimento TPV]
                                ) * 100,

                            "Resultado - Fat. Líquido",
                                CALCULATE (
                                    'BASE'[% Atingimento Faturamento Líquido]
                                ) * 100,

                            "Consistência - BSC",
                                CALCULATE (
                                    'BASE'[% Atingimento BSC]
                                ) * 100,

                            "Consistência - Encanta/NPS",
                                CALCULATE (
                                    'BASE'[% Atingimento Encanta BRA]
                                ) * 100,

                            "PDD",
                                CALCULATE (
                                    'BASE'[% Atingimento PDD]
                                ) * 100
                        )
                ),

                "@IndicadorL",
                    _Nome,

                "@OrdemL",
                    _Ordem,

                "@CorL",
                    _Cor,

                "@PosL",
                    [@Pos],

                "@SortL",
                    [@Sort],

                "@AnoL",
                    VALUE ( 'BASE'[ANO] ),

                "@MesNumL",
                    [@MesNum],

                "@DataRefL",
                    [@DataRef],

                "@ValorL",
                    [@ValorTemp],

                /*
                   Valor visual.

                   O dado REAL continua em @ValorL.

                   Só a posição gráfica é limitada a ±200%.
                */

                "@ValorPlotL",

                    VAR _V =
                        [@ValorTemp]

                    RETURN

                        IF (
                            ISBLANK ( _V ),
                            BLANK (),
                            MAX (
                                -200,
                                MIN (
                                    200,
                                    _V
                                )
                            )
                        )
            )
    )


VAR _DadosValidos =
    FILTER (
        _DadosLongos,
        NOT ISBLANK ( [@ValorL] )
    )


VAR _TemDados =
    COUNTROWS (
        _DadosValidos
    ) > 0


/* ============================================================
   ESCALA Y

   IMPORTANTE:
   usa @ValorPlotL, portanto um 900% não distorce o gráfico.
   ============================================================ */

VAR _MinData =
    MINX (
        _DadosValidos,
        [@ValorPlotL]
    )

VAR _MaxData =
    MAXX (
        _DadosValidos,
        [@ValorPlotL]
    )


VAR _MinBase =
    MIN (
        _MinData,
        100
    )

VAR _MaxBase =
    MAX (
        _MaxData,
        100
    )


VAR _Amplitude =
    MAX (
        15,
        _MaxBase - _MinBase
    )


VAR _YMin =
    IF (
        _MinBase < 0,

        _MinBase
            - _Amplitude * 0.10,

        MAX (
            0,
            _MinBase
                - _Amplitude * 0.18
        )
    )


VAR _YMax =
    _MaxBase
        + _Amplitude * 0.20


VAR _YRange =
    MAX (
        1,
        _YMax - _YMin
    )


/* ============================================================
   SÉRIE ÚNICA / INSIGHTS
   ============================================================ */

VAR _SerieUnica =
    FILTER (
        _DadosValidos,
        [@IndicadorL] = _IndicadorUnico
    )


VAR _MediaAtingimento =
    IF (
        _QtdIndicadores = 1,
        AVERAGEX (
            _SerieUnica,
            [@ValorL]
        )
    )


VAR _UltimoSort =
    MAXX (
        _SerieUnica,
        [@SortL]
    )


VAR _UltimoValor =
    MAXX (
        FILTER (
            _SerieUnica,
            [@SortL] = _UltimoSort
        ),
        [@ValorL]
    )


/* último período anterior disponível */

VAR _SortAnterior =
    MAXX (
        FILTER (
            _SerieUnica,
            [@SortL] < _UltimoSort
        ),
        [@SortL]
    )


VAR _ValorAnterior =
    MAXX (
        FILTER (
            _SerieUnica,
            [@SortL] = _SortAnterior
        ),
        [@ValorL]
    )


VAR _MoM =
    IF (
        _QtdIndicadores = 1
            && NOT ISBLANK ( _ValorAnterior )
            && _ValorAnterior <> 0,

        DIVIDE (
            _UltimoValor - _ValorAnterior,
            ABS ( _ValorAnterior )
        ) * 100
    )


/* ============================================================
   YoY
   mesmo mês do ano anterior
   ============================================================ */

VAR _ValorAnoAnterior =
    MAXX (
        FILTER (
            _SerieUnica,
            [@SortL] = _UltimoSort - 100
        ),
        [@ValorL]
    )


VAR _YoY =
    IF (
        _QtdIndicadores = 1
            && NOT ISBLANK ( _ValorAnoAnterior )
            && _ValorAnoAnterior <> 0,

        DIVIDE (
            _UltimoValor - _ValorAnoAnterior,
            ABS ( _ValorAnoAnterior )
        ) * 100
    )


/* ============================================================
   % DOS MESES >= 100%
   ============================================================ */

VAR _QtdMesesMeta =
    COUNTROWS (
        FILTER (
            _SerieUnica,
            [@ValorL] >= 100
        )
    )


VAR _PctMesesMeta =
    IF (
        _QtdIndicadores = 1,

        DIVIDE (
            _QtdMesesMeta,
            COUNTROWS ( _SerieUnica )
        ) * 100
    )


/* ============================================================
   MÁXIMO / MÍNIMO
   ============================================================ */

VAR _ValorMax =
    IF (
        _QtdIndicadores = 1,
        MAXX (
            _SerieUnica,
            [@ValorL]
        )
    )


VAR _ValorMin =
    IF (
        _QtdIndicadores = 1,
        MINX (
            _SerieUnica,
            [@ValorL]
        )
    )


VAR _SortMax =
    MINX (
        FILTER (
            _SerieUnica,
            [@ValorL] = _ValorMax
        ),
        [@SortL]
    )


VAR _SortMin =
    MINX (
        FILTER (
            _SerieUnica,
            [@ValorL] = _ValorMin
        ),
        [@SortL]
    )


/* ============================================================
   INSIGHTS DO CABEÇALHO
   ============================================================ */

VAR _CorMoM =
    IF (
        _MoM >= 0,
        _CorBest,
        _CorWorst
    )


VAR _CorYoY =
    IF (
        _YoY >= 0,
        _CorBest,
        _CorWorst
    )


VAR _Insights =
    IF (
        _QtdIndicadores = 1,

        "
        <div class='insights'>

            <div class='insight-card'>
                <div class='insight-label'>ATINGIMENTO MÉDIO</div>
                <div class='insight-value'>"
                    &
                    FORMAT (
                        _MediaAtingimento,
                        "0.0"
                    )
                    &
                    "%
                </div>
            </div>


            <div class='insight-card'>
                <div class='insight-label'>MOM</div>
                <div
                    class='insight-value'
                    style='color:"
                        &
                        _CorMoM
                        &
                    ";'
                >"
                    &
                    IF (
                        ISBLANK ( _MoM ),
                        "—",
                        IF (
                            _MoM >= 0,
                            "+",
                            ""
                        )
                            &
                            FORMAT (
                                _MoM,
                                "0.0"
                            )
                            &
                            "%"
                    )
                    &
                "</div>

                <div class='insight-desc'>
                    vs mês anterior
                </div>
            </div>


            <div class='insight-card'>
                <div class='insight-label'>YOY</div>
                <div
                    class='insight-value'
                    style='color:"
                        &
                        _CorYoY
                        &
                    ";'
                >"
                    &
                    IF (
                        ISBLANK ( _YoY ),
                        "—",
                        IF (
                            _YoY >= 0,
                            "+",
                            ""
                        )
                            &
                            FORMAT (
                                _YoY,
                                "0.0"
                            )
                            &
                            "%"
                    )
                    &
                "</div>

                <div class='insight-desc'>
                    vs mesmo mês ano anterior
                </div>
            </div>


            <div class='insight-card'>
                <div class='insight-label'>MESES NA META</div>

                <div class='insight-value'>"
                    &
                    FORMAT (
                        _PctMesesMeta,
                        "0"
                    )
                    &
                    "%
                </div>

                <div class='insight-desc'>
                    atingimento ≥ 100%
                </div>
            </div>

        </div>
        "
    )


/* ============================================================
   GRID
   ============================================================ */

VAR _Grid =
    CONCATENATEX (
        GENERATESERIES (
            0,
            3,
            1
        ),

        VAR _I =
            [Value]

        VAR _V =
            _YMin
                + DIVIDE (
                    _YRange,
                    3
                ) * _I

        VAR _Y =
            _PadTop
                + _ChartH
                - DIVIDE (
                    _V - _YMin,
                    _YRange
                ) * _ChartH

        RETURN

            "
            <line
                x1='" & FORMAT ( _PadLeft, "0.0", "en-US" ) & "'
                x2='" & FORMAT ( _W - _PadRight, "0.0", "en-US" ) & "'
                y1='" & FORMAT ( _Y, "0.0", "en-US" ) & "'
                y2='" & FORMAT ( _Y, "0.0", "en-US" ) & "'
                stroke='" & _CorGrid & "'
                stroke-width='1'
                stroke-dasharray='2 4'
            />
            ",

        ""
    )


/* ============================================================
   LINHA META
   ============================================================ */

VAR _Y100 =
    _PadTop
        + _ChartH
        - DIVIDE (
            100 - _YMin,
            _YRange
        ) * _ChartH


VAR _LinhaMeta =
    "
    <line
        x1='" & FORMAT ( _PadLeft, "0", "en-US" ) & "'
        x2='" & FORMAT ( _W - _PadRight, "0", "en-US" ) & "'
        y1='" & FORMAT ( _Y100, "0.0", "en-US" ) & "'
        y2='" & FORMAT ( _Y100, "0.0", "en-US" ) & "'
        stroke='" & _CorMeta & "'
        stroke-width='1.7'
        stroke-dasharray='7 6'
        opacity='.9'
    />

    <text
        x='" & FORMAT ( _PadLeft - 12, "0", "en-US" ) & "'
        y='" & FORMAT ( _Y100 + 4, "0.0", "en-US" ) & "'
        text-anchor='end'
        font-family='Segoe UI,Arial,sans-serif'
        font-size='11'
        font-weight='700'
        fill='" & _CorMeta & "'
    >
        100%
    </text>
    "


/* ============================================================
   INÍCIO DA FUNÇÃO
   ============================================================ */

VAR _MarcadorFuncao =
    IF (
        _DataFuncaoDentro
            && NOT ISBLANK ( _PosFuncao ),

        VAR _X =
            _PadLeft
                + DIVIDE (
                    _PosFuncao - 1,
                    _DivX
                ) * _ChartW


        VAR _BoxW =
            170

        VAR _BoxH =
            40


        VAR _BoxXOriginal =
            _X - DIVIDE ( _BoxW, 2 )


        VAR _BoxX =
            MAX (
                _PadLeft,
                MIN (
                    _W - _PadRight - _BoxW,
                    _BoxXOriginal
                )
            )


        VAR _BoxCenter =
            _BoxX
                + DIVIDE ( _BoxW, 2 )


        RETURN

            "
            <line
                x1='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                x2='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                y1='" & FORMAT ( _PadTop + 4, "0", "en-US" ) & "'
                y2='" & FORMAT ( _PadTop + _ChartH, "0", "en-US" ) & "'
                stroke='" & _CorFuncao & "'
                stroke-width='1.7'
                stroke-dasharray='5 5'
                opacity='.75'
            />


            <circle
                cx='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                cy='" & FORMAT ( _PadTop + 4, "0", "en-US" ) & "'
                r='4.5'
                fill='" & _CorFuncao & "'
            />


            <rect
                x='" & FORMAT ( _BoxX, "0.0", "en-US" ) & "'
                y='4'
                width='" & FORMAT ( _BoxW, "0", "en-US" ) & "'
                height='" & FORMAT ( _BoxH, "0", "en-US" ) & "'
                rx='9'
                fill='#FFF7F1'
                stroke='" & _CorFuncao & "'
                stroke-width='1'
            />


            <text
                x='" & FORMAT ( _BoxCenter, "0.0", "en-US" ) & "'
                y='20'
                text-anchor='middle'
                font-family='Segoe UI,Arial,sans-serif'
                font-size='12'
                font-weight='700'
                fill='" & _CorFuncao & "'
            >
                Início da função
            </text>


            <text
                x='" & FORMAT ( _BoxCenter, "0.0", "en-US" ) & "'
                y='35'
                text-anchor='middle'
                font-family='Segoe UI,Arial,sans-serif'
                font-size='11'
                font-weight='500'
                fill='" & _CorSub & "'
            >
                "
                    &
                    FORMAT (
                        _DataFuncao,
                        "dd/MM/yyyy"
                    )
                    &
                "
            </text>
            "
    )


/* ============================================================
   EIXO X
   ============================================================ */

VAR _MesesSVG =
    CONCATENATEX (
        _Periodos,

        VAR _X =
            _PadLeft
                + DIVIDE (
                    [@Pos] - 1,
                    _DivX
                ) * _ChartW

        VAR _Mes =
            SWITCH (
                [@MesNum],
                1, "jan",
                2, "fev",
                3, "mar",
                4, "abr",
                5, "mai",
                6, "jun",
                7, "jul",
                8, "ago",
                9, "set",
                10, "out",
                11, "nov",
                12, "dez"
            )

        RETURN

            "
            <text
                x='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                y='" & FORMAT ( _PadTop + _ChartH + 24, "0", "en-US" ) & "'
                text-anchor='middle'
                font-family='Segoe UI,Arial,sans-serif'
                font-size='12'
                font-weight='500'
                fill='" & _CorSub & "'
            >
                " & _Mes & "
            </text>
            ",

        "",
        [@Sort],
        ASC
    )


/* ============================================================
   ANOS
   ============================================================ */

VAR _Anos =
    DISTINCT (
        SELECTCOLUMNS (
            _Periodos,
            "@Ano",
                VALUE ( 'BASE'[ANO] )
        )
    )


VAR _AnosSVG =
    CONCATENATEX (
        _Anos,

        VAR _AnoAtual =
            [@Ano]

        VAR _Inicio =
            MINX (
                FILTER (
                    _Periodos,
                    VALUE ( 'BASE'[ANO] ) = _AnoAtual
                ),
                [@Pos]
            )

        VAR _Fim =
            MAXX (
                FILTER (
                    _Periodos,
                    VALUE ( 'BASE'[ANO] ) = _AnoAtual
                ),
                [@Pos]
            )

        VAR _X1 =
            _PadLeft
                + DIVIDE (
                    _Inicio - 1,
                    _DivX
                ) * _ChartW

        VAR _X2 =
            _PadLeft
                + DIVIDE (
                    _Fim - 1,
                    _DivX
                ) * _ChartW

        VAR _XC =
            DIVIDE (
                _X1 + _X2,
                2
            )

        RETURN

            "
            <text
                x='" & FORMAT ( _XC, "0.0", "en-US" ) & "'
                y='" & FORMAT ( _PadTop + _ChartH + 50, "0", "en-US" ) & "'
                text-anchor='middle'
                font-family='Segoe UI,Arial,sans-serif'
                font-size='13'
                font-weight='700'
                fill='" & _CorTexto & "'
            >
                " & FORMAT ( _AnoAtual, "0" ) & "
            </text>
            ",

        ""
    )


/* ============================================================
   LEGENDA
   ============================================================ */

VAR _Legenda =
    CONCATENATEX (
        _IndicadoresSelecionados,

        VAR _Nome =
            [@Indicador]

        VAR _Cor =
            SWITCH (
                _Nome,
                "Resultado - ROB Crédito", _CorROB_Credito,
                "Resultado - ROB Não crédito", _CorROB_NaoCredito,
                "Resultado - TPV", _CorTPV,
                "Resultado - Fat. Líquido", _CorFatLiquido,
                "Consistência - BSC", _CorBSC,
                "Consistência - Encanta/NPS", _CorEncanta,
                "PDD", _CorPDD,
                "#777777"
            )

        RETURN

            "
            <div class='legend-item'>
                <span
                    class='legend-line'
                    style='border-color:" & _Cor & ";'
                ></span>

                <span>"
                    & _Nome &
                "</span>
            </div>
            ",

        "",
        [@Ordem],
        ASC
    )


/* ============================================================
   SÉRIES
   ============================================================ */

VAR _SeriesSVG =
    CONCATENATEX (
        _IndicadoresSelecionados,

        VAR _Nome =
            [@Indicador]

        VAR _Ordem =
            [@Ordem]

        VAR _Cor =
            SWITCH (
                _Nome,
                "Resultado - ROB Crédito", _CorROB_Credito,
                "Resultado - ROB Não crédito", _CorROB_NaoCredito,
                "Resultado - TPV", _CorTPV,
                "Resultado - Fat. Líquido", _CorFatLiquido,
                "Consistência - BSC", _CorBSC,
                "Consistência - Encanta/NPS", _CorEncanta,
                "PDD", _CorPDD,
                "#777777"
            )


        VAR _Serie =
            FILTER (
                _DadosValidos,
                [@IndicadorL] = _Nome
            )


        /* ====================================================
           OFFSET DOS LABELS
           ==================================================== */

        VAR _Offset =
            SWITCH (
                MOD ( _Ordem, 6 ),
                0, -16,
                1, 22,
                2, -33,
                3, 39,
                4, -49,
                5, 55,
                -16
            )


        /* ====================================================
           PATH SUAVE EM BÉZIER
           ==================================================== */

        VAR _Path =
            CONCATENATEX (
                _Serie,

                VAR _Pos =
                    [@PosL]
