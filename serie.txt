Histórico Indicadores HTML =
VAR _W = 1400
VAR _H = 500

VAR _PadLeft = 48
VAR _PadRight = 42
VAR _PadTop = 82
VAR _PadBottom = 55

VAR _ChartW = _W - _PadLeft - _PadRight
VAR _ChartH = _H - _PadTop - _PadBottom


/* ============================================================
   CORES / DESIGN
   Inspirado no visual original
   ============================================================ */

VAR _ColorInk = "#3D2F28"
VAR _ColorSub = "#6B5D52"
VAR _ColorMuted = "#737378"
VAR _ColorGrid = "#EDE9E3"

VAR _ColorBest = "#4A7A3B"
VAR _ColorWorst = "#A8524B"
VAR _ColorPrimary = "#C96A3A"


/* ============================================================
   FIELD PARAMETER

   Ajustado para os nomes em português do seu modelo:
   Parâmetro
   Parâmetro Campos
   Parâmetro Pedido
   ============================================================ */

VAR _IndicadoresSelecionados =
    SELECTCOLUMNS (
        SUMMARIZE (
            'Parâmetro',
            'Parâmetro'[Parâmetro],
            'Parâmetro'[Parâmetro Campos],
            'Parâmetro'[Parâmetro Pedido]
        ),
        "@Indicador",
            'Parâmetro'[Parâmetro],

        "@Ordem",
            'Parâmetro'[Parâmetro Pedido]
    )


VAR _QtdIndicadores =
    COUNTROWS ( _IndicadoresSelecionados )


/* ============================================================
   PERÍODOS
   ============================================================ */

VAR _PeriodosBase =
    CALCULATETABLE (
        SUMMARIZE (
            'BASE',
            'BASE'[ANO],
            'BASE'[MESANO]
        ),
        ALLSELECTED (
            'BASE'[ANO],
            'BASE'[MESANO]
        )
    )


/* ============================================================
   CONVERTE MESANO EM NÚMERO

   Funciona com:
   janeiro
   jan
   janeiro/2025
   jan/2025

   etc.
   ============================================================ */

VAR _PeriodosComMes =
    ADDCOLUMNS (
        _PeriodosBase,

        "@MesTexto",
            LOWER (
                TRIM (
                    FORMAT (
                        'BASE'[MESANO],
                        ""
                    )
                )
            ),

        "@MesNum",

            VAR _M =
                LOWER (
                    TRIM (
                        FORMAT (
                            'BASE'[MESANO],
                            ""
                        )
                    )
                )

            RETURN

                SWITCH (
                    TRUE (),

                    LEFT ( _M, 3 ) = "jan", 1,
                    LEFT ( _M, 3 ) = "fev", 2,
                    LEFT ( _M, 3 ) = "mar", 3,
                    LEFT ( _M, 3 ) = "abr", 4,
                    LEFT ( _M, 3 ) = "mai", 5,
                    LEFT ( _M, 3 ) = "jun", 6,
                    LEFT ( _M, 3 ) = "jul", 7,
                    LEFT ( _M, 3 ) = "ago", 8,
                    LEFT ( _M, 3 ) = "set", 9,
                    LEFT ( _M, 3 ) = "out", 10,
                    LEFT ( _M, 3 ) = "nov", 11,
                    LEFT ( _M, 3 ) = "dez", 12,

                    MONTH (
                        DATEVALUE (
                            "01/" &
                            FORMAT ( 'BASE'[MESANO], "" )
                        )
                    )
                )
    )


VAR _PeriodosSort =
    ADDCOLUMNS (
        _PeriodosComMes,

        "@Sort",
            'BASE'[ANO] * 100
                + [@MesNum]
    )


VAR _Periodos =
    ADDCOLUMNS (
        _PeriodosSort,

        "@Pos",
            VAR _SortAtual =
                [@Sort]

            RETURN

                RANKX (
                    _PeriodosSort,
                    [@Sort],
                    _SortAtual,
                    ASC,
                    DENSE
                )
    )


VAR _QtdPeriodos =
    COUNTROWS ( _Periodos )

VAR _DivX =
    MAX (
        1,
        _QtdPeriodos - 1
    )


/* ============================================================
   TABELA LONGA
   ============================================================ */

VAR _DadosLongos =
    GENERATE (
        _IndicadoresSelecionados,

        VAR _NomeIndicador =
            [@Indicador]

        VAR _OrdemIndicador =
            [@Ordem]

        VAR _Cor =
            SWITCH (
                _NomeIndicador,

                "Resultado - ROB Crédito",
                    "#C96A3A",

                "Resultado - ROB Não crédito",
                    "#6B5D52",

                "Resultado - TPV",
                    "#B68A67",

                "Resultado - Fat. Líquido",
                    "#D19A66",

                "Consistência - Encanta/NPS",
                    "#4A7A3B",

                "Consistência - BSC",
                    "#78975E",

                "PDD",
                    "#737378",

                "#6B5D52"
            )

        RETURN

            SELECTCOLUMNS (
                ADDCOLUMNS (
                    _Periodos,

                    "@ValorTemp",

                        SWITCH (
                            _NomeIndicador,

                            "Resultado - ROB Crédito",
                                CALCULATE (
                                    'BASE'[% Atingimento ROB Crédito]
                                ) * 100,

                            "Resultado - ROB Não crédito",
                                CALCULATE (
                                    'BASE'[% Atingimento ROB Não Crédito]
                                ) * 100,

                            "Resultado - TPV",
                                CALCULATE (
                                    'BASE'[% Atingimento TPV]
                                ) * 100,

                            "Resultado - Fat. Líquido",
                                CALCULATE (
                                    'BASE'[% Atingimento Faturamento Líquido]
                                ) * 100,

                            "Consistência - Encanta/NPS",
                                CALCULATE (
                                    'BASE'[% Atingimento Encanta BRA]
                                ) * 100,

                            "Consistência - BSC",
                                CALCULATE (
                                    'BASE'[% Atingimento BSC]
                                ) * 100,

                            "PDD",
                                CALCULATE (
                                    'BASE'[% Atingimento PDD]
                                ) * 100
                        )
                ),

                "@IndicadorL",
                    _NomeIndicador,

                "@OrdemL",
                    _OrdemIndicador,

                "@CorL",
                    _Cor,

                "@PosL",
                    [@Pos],

                "@SortL",
                    [@Sort],

                "@AnoL",
                    'BASE'[ANO],

                "@MesNumL",
                    [@MesNum],

                "@MesAnoL",
                    'BASE'[MESANO],

                "@ValorL",
                    [@ValorTemp]
            )
    )


/* ============================================================
   ESCALA Y
   ============================================================ */

VAR _DadosValidos =
    FILTER (
        _DadosLongos,
        NOT ISBLANK ( [@ValorL] )
    )


VAR _MinData =
    MINX (
        _DadosValidos,
        [@ValorL]
    )


VAR _MaxData =
    MAXX (
        _DadosValidos,
        [@ValorL]
    )


VAR _MinBase =
    MIN (
        _MinData,
        100
    )


VAR _MaxBase =
    MAX (
        _MaxData,
        100
    )


VAR _Amplitude =
    MAX (
        10,
        _MaxBase - _MinBase
    )


VAR _YMin =
    MAX (
        0,
        _MinBase - _Amplitude * 0.15
    )


VAR _YMax =
    _MaxBase + _Amplitude * 0.18


VAR _YRange =
    MAX (
        1,
        _YMax - _YMin
    )


/* ============================================================
   PRIMEIRO / ÚLTIMO PERÍODO
   ============================================================ */

VAR _PrimeiraPos =
    MINX (
        _Periodos,
        [@Pos]
    )


VAR _UltimaPos =
    MAXX (
        _Periodos,
        [@Pos]
    )


VAR _PrimeiroPeriodo =
    MAXX (
        FILTER (
            _Periodos,
            [@Pos] = _PrimeiraPos
        ),
        FORMAT (
            'BASE'[MESANO],
            ""
        )
            & "/"
            & FORMAT (
                'BASE'[ANO],
                "0"
            )
    )


VAR _UltimoPeriodo =
    MAXX (
        FILTER (
            _Periodos,
            [@Pos] = _UltimaPos
        ),
        FORMAT (
            'BASE'[MESANO],
            ""
        )
            & "/"
            & FORMAT (
                'BASE'[ANO],
                "0"
            )
    )


/* ============================================================
   RESUMO DE EVOLUÇÃO

   Quando existe somente 1 indicador selecionado,
   mostra inicial → final igual ao conceito do original.

   Com vários indicadores, mantém cabeçalho mais limpo.
   ============================================================ */

VAR _IndicadorUnico =
    IF (
        _QtdIndicadores = 1,
        MAXX (
            _IndicadoresSelecionados,
            [@Indicador]
        )
    )


VAR _ValorInicial =
    IF (
        _QtdIndicadores = 1,

        MAXX (
            FILTER (
                _DadosValidos,
                [@IndicadorL] = _IndicadorUnico
                    && [@PosL] = _PrimeiraPos
            ),
            [@ValorL]
        )
    )


VAR _ValorFinal =
    IF (
        _QtdIndicadores = 1,

        MAXX (
            FILTER (
                _DadosValidos,
                [@IndicadorL] = _IndicadorUnico
                    && [@PosL] = _UltimaPos
            ),
            [@ValorL]
        )
    )


VAR _Variacao =
    IF (
        _QtdIndicadores = 1
            && NOT ISBLANK ( _ValorInicial )
            && _ValorInicial <> 0,

        DIVIDE (
            _ValorFinal - _ValorInicial,
            ABS ( _ValorInicial )
        ) * 100
    )


VAR _VariacaoPositiva =
    _Variacao >= 0


VAR _CorVariacao =
    IF (
        _VariacaoPositiva,
        _ColorBest,
        _ColorWorst
    )


VAR _HeaderResumo =
    IF (
        _QtdIndicadores = 1,

        "
        <div class='summary-row'>

            <span
                class='growth-pill'
                style='
                    color:" & _CorVariacao & ";
                    background:"
                        &
                        IF (
                            _VariacaoPositiva,
                            "rgba(74,122,59,.10)",
                            "rgba(168,82,75,.10)"
                        )
                        &
                    ";
                '
            >
                "
                    &
                    IF (
                        _VariacaoPositiva,
                        "↑ ",
                        "↓ "
                    )
                    &
                    IF (
                        _VariacaoPositiva,
                        "+",
                        ""
                    )
                    &
                    FORMAT (
                        _Variacao,
                        "0.0"
                    )
                    &
                    "%
            </span>

            <span class='summary-value'>
                "
                    &
                    FORMAT (
                        _ValorInicial,
                        "0.#"
                    )
                    &
                    "%
            </span>

            <span class='arrow'>
                →
            </span>

            <span class='summary-value end'>
                "
                    &
                    FORMAT (
                        _ValorFinal,
                        "0.#"
                    )
                    &
                    "%
            </span>

            <span class='period-range'>
                ("
                    &
                    _PrimeiroPeriodo
                    &
                    " — "
                    &
                    _UltimoPeriodo
                    &
                ")
            </span>

        </div>
        ",

        "
        <div class='summary-row'>
            <span class='period-range'>
                "
                    &
                    _PrimeiroPeriodo
                    &
                    " — "
                    &
                    _UltimoPeriodo
                    &
                    " · "
                    &
                    FORMAT (
                        _QtdIndicadores,
                        "0"
                    )
                    &
                    " indicadores selecionados
            </span>
        </div>
        "
    )


/* ============================================================
   GRID
   ============================================================ */

VAR _Grid =
    CONCATENATEX (
        GENERATESERIES (
            0,
            3,
            1
        ),

        VAR _I =
            [Value]

        VAR _V =
            _YMin
                + DIVIDE (
                    _YRange,
                    3
                ) * _I

        VAR _Y =
            _PadTop
                + _ChartH
                - DIVIDE (
                    _V - _YMin,
                    _YRange
                ) * _ChartH

        RETURN

            "
            <line
                x1='" & FORMAT ( _PadLeft - 8, "0.0", "en-US" ) & "'
                x2='" & FORMAT ( _W - _PadRight + 8, "0.0", "en-US" ) & "'
                y1='" & FORMAT ( _Y, "0.0", "en-US" ) & "'
                y2='" & FORMAT ( _Y, "0.0", "en-US" ) & "'
                stroke='" & _ColorGrid & "'
                stroke-width='0.7'
            />

            <text
                x='" & FORMAT ( _PadLeft - 13, "0.0", "en-US" ) & "'
                y='" & FORMAT ( _Y + 3, "0.0", "en-US" ) & "'
                text-anchor='end'
                font-family='Jost,Segoe UI,sans-serif'
                font-size='8'
                fill='" & _ColorMuted & "'
            >
                " & FORMAT ( _V, "0" ) & "%
            </text>
            ",

        ""
    )


/* ============================================================
   LINHA 100%
   ============================================================ */

VAR _Y100 =
    _PadTop
        + _ChartH
        - DIVIDE (
            100 - _YMin,
            _YRange
        ) * _ChartH


VAR _Linha100 =
    "
    <line
        x1='" & FORMAT ( _PadLeft - 8, "0", "en-US" ) & "'
        x2='" & FORMAT ( _W - _PadRight + 8, "0", "en-US" ) & "'
        y1='" & FORMAT ( _Y100, "0.0", "en-US" ) & "'
        y2='" & FORMAT ( _Y100, "0.0", "en-US" ) & "'
        stroke='" & _ColorPrimary & "'
        stroke-width='0.8'
        stroke-dasharray='4 4'
        opacity='.38'
    />

    <text
        x='" & FORMAT ( _PadLeft - 13, "0", "en-US" ) & "'
        y='" & FORMAT ( _Y100 - 5, "0.0", "en-US" ) & "'
        text-anchor='end'
        font-family='Jost,Segoe UI,sans-serif'
        font-size='7'
        font-weight='500'
        fill='" & _ColorPrimary & "'
        opacity='.7'
    >
        META
    </text>
    "


/* ============================================================
   EIXO X

   Agora ordenado por @Sort:
   202501, 202502, 202503...
   ============================================================ */

VAR _MesesSVG =
    CONCATENATEX (
        _Periodos,

        VAR _Pos =
            [@Pos]

        VAR _X =
            _PadLeft
                + DIVIDE (
                    _Pos - 1,
                    _DivX
                ) * _ChartW

        VAR _MesNum =
            [@MesNum]

        VAR _MesAbrev =
            SWITCH (
                _MesNum,
                1, "jan",
                2, "fev",
                3, "mar",
                4, "abr",
                5, "mai",
                6, "jun",
                7, "jul",
                8, "ago",
                9, "set",
                10, "out",
                11, "nov",
                12, "dez"
            )

        RETURN

            "
            <text
                x='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                y='" & FORMAT ( _PadTop + _ChartH + 19, "0", "en-US" ) & "'
                text-anchor='middle'
                font-family='Jost,Segoe UI,sans-serif'
                font-size='10'
                font-weight='400'
                fill='" & _ColorMuted & "'
            >
                " & _MesAbrev & "
            </text>
            ",

        "",
        [@Sort],
        ASC
    )


/* ============================================================
   ANOS AGRUPADOS
   ============================================================ */

VAR _Anos =
    DISTINCT (
        SELECTCOLUMNS (
            _Periodos,
            "@Ano",
                'BASE'[ANO]
        )
    )


VAR _AnosSVG =
    CONCATENATEX (
        _Anos,

        VAR _AnoAtual =
            [@Ano]

        VAR _Inicio =
            MINX (
                FILTER (
                    _Periodos,
                    'BASE'[ANO] = _AnoAtual
                ),
                [@Pos]
            )

        VAR _Fim =
            MAXX (
                FILTER (
                    _Periodos,
                    'BASE'[ANO] = _AnoAtual
                ),
                [@Pos]
            )

        VAR _X1 =
            _PadLeft
                + DIVIDE (
                    _Inicio - 1,
                    _DivX
                ) * _ChartW

        VAR _X2 =
            _PadLeft
                + DIVIDE (
                    _Fim - 1,
                    _DivX
                ) * _ChartW

        VAR _XC =
            DIVIDE (
                _X1 + _X2,
                2
            )

        RETURN

            "
            <text
                x='" & FORMAT ( _XC, "0.0", "en-US" ) & "'
                y='" & FORMAT ( _PadTop + _ChartH + 41, "0", "en-US" ) & "'
                text-anchor='middle'
                font-family='Jost,Segoe UI,sans-serif'
                font-size='9'
                font-weight='500'
                letter-spacing='1'
                fill='" & _ColorSub & "'
            >
                " & FORMAT ( _AnoAtual, "0" ) & "
            </text>
            ",

        ""
    )


/* ============================================================
   SEPARAÇÃO DE ANOS
   ============================================================ */

VAR _AnoMin =
    MINX (
        _Anos,
        [@Ano]
    )


VAR _SeparadoresAno =
    CONCATENATEX (
        FILTER (
            _Anos,
            [@Ano] <> _AnoMin
        ),

        VAR _AnoAtual =
            [@Ano]

        VAR _Primeira =
            MINX (
                FILTER (
                    _Periodos,
                    'BASE'[ANO] = _AnoAtual
                ),
                [@Pos]
            )

        VAR _XA =
            _PadLeft
                + DIVIDE (
                    _Primeira - 1,
                    _DivX
                ) * _ChartW

        VAR _XB =
            _PadLeft
                + DIVIDE (
                    _Primeira - 2,
                    _DivX
                ) * _ChartW

        VAR _X =
            DIVIDE (
                _XA + _XB,
                2
            )

        RETURN

            "
            <line
                x1='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                x2='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                y1='" & FORMAT ( _PadTop, "0", "en-US" ) & "'
                y2='" & FORMAT ( _PadTop + _ChartH + 30, "0", "en-US" ) & "'
                stroke='#DDD8D2'
                stroke-width='.7'
                stroke-dasharray='2 4'
            />
            ",

        ""
    )


/* ============================================================
   LEGENDA
   ============================================================ */

VAR _Legenda =
    CONCATENATEX (
        _IndicadoresSelecionados,

        VAR _Nome =
            [@Indicador]

        VAR _Cor =
            SWITCH (
                _Nome,

                "Resultado - ROB Crédito",
                    "#C96A3A",

                "Resultado - ROB Não crédito",
                    "#6B5D52",

                "Resultado - TPV",
                    "#B68A67",

                "Resultado - Fat. Líquido",
                    "#D19A66",

                "Consistência - Encanta/NPS",
                    "#4A7A3B",

                "Consistência - BSC",
                    "#78975E",

                "PDD",
                    "#737378",

                "#6B5D52"
            )

        RETURN

            "
            <div class='legend-item'>

                <span
                    class='legend-line'
                    style='background:" & _Cor & ";'
                ></span>

                <span>"
                    & _Nome &
                "</span>

            </div>
            ",

        "",
        [@Ordem],
        ASC
    )


/* ============================================================
   SÉRIES
   ============================================================ */

VAR _SeriesSVG =
    CONCATENATEX (
        _IndicadoresSelecionados,

        VAR _Nome =
            [@Indicador]

        VAR _Ordem =
            [@Ordem]

        VAR _Cor =
            SWITCH (
                _Nome,

                "Resultado - ROB Crédito",
                    "#C96A3A",

                "Resultado - ROB Não crédito",
                    "#6B5D52",

                "Resultado - TPV",
                    "#B68A67",

                "Resultado - Fat. Líquido",
                    "#D19A66",

                "Consistência - Encanta/NPS",
                    "#4A7A3B",

                "Consistência - BSC",
                    "#78975E",

                "PDD",
                    "#737378",

                "#6B5D52"
            )


        VAR _Serie =
            FILTER (
                _DadosValidos,
                [@IndicadorL] = _Nome
            )


        VAR _Points =
            CONCATENATEX (
                _Serie,

                VAR _Pos =
                    [@PosL]

                VAR _Valor =
                    [@ValorL]

                VAR _X =
                    _PadLeft
                        + DIVIDE (
                            _Pos - 1,
                            _DivX
                        ) * _ChartW

                VAR _Y =
                    _PadTop
                        + _ChartH
                        - DIVIDE (
                            _Valor - _YMin,
                            _YRange
                        ) * _ChartH

                RETURN

                    FORMAT (
                        _X,
                        "0.0",
                        "en-US"
                    )
                        & ","
                        &
                    FORMAT (
                        _Y,
                        "0.0",
                        "en-US"
                    ),

                " ",
                [@SortL],
                ASC
            )


        /* Maior e menor valor de cada série */

        VAR _MaiorValor =
            MAXX (
                _Serie,
                [@ValorL]
            )

        VAR _MenorValor =
            MINX (
                _Serie,
                [@ValorL]
            )


        VAR _Detalhes =
            CONCATENATEX (
                _Serie,

                VAR _Pos =
                    [@PosL]

                VAR _Valor =
                    [@ValorL]

                VAR _X =
                    _PadLeft
                        + DIVIDE (
                            _Pos - 1,
                            _DivX
                        ) * _ChartW

                VAR _Y =
                    _PadTop
                        + _ChartH
                        - DIVIDE (
                            _Valor - _YMin,
                            _YRange
                        ) * _ChartH


                VAR _EhMaior =
                    _Valor = _MaiorValor


                VAR _EhMenor =
                    _Valor = _MenorValor
                        && _MenorValor <> _MaiorValor


                /*
                   Com 1 série:
                   mostra labels em todos os pontos.

                   Com várias:
                   mostra apenas peak/low para não poluir.
                */

                VAR _MostrarValor =
                    _QtdIndicadores = 1
                        || _EhMaior
                        || _EhMenor


                VAR _CorDestaque =
                    IF (
                        _EhMaior,
                        _ColorBest,
                        IF (
                            _EhMenor,
                            _ColorWorst,
                            _Cor
                        )
                    )


                VAR _LabelEspecial =
                    IF (
                        _EhMaior,
                        "PICO",
                        IF (
                            _EhMenor,
                            "MENOR",
                            ""
                        )
                    )


                RETURN

                    "
                    <circle
                        cx='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                        cy='" & FORMAT ( _Y, "0.0", "en-US" ) & "'
                        r='"
                            &
                            IF (
                                _EhMaior || _EhMenor,
                                "3.5",
                                "2.4"
                            )
                            &
                        "'
                        fill='" & _CorDestaque & "'
                        stroke='#FFFFFF'
                        stroke-width='1'
                    />
                    "

                    &

                    IF (
                        _MostrarValor,

                        "
                        <text
                            x='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                            y='" & FORMAT ( _Y - 12, "0.0", "en-US" ) & "'
                            text-anchor='middle'
                            font-family='Jost,Segoe UI,sans-serif'
                            font-size='9'
                            font-weight='"
                                &
                                IF (
                                    _EhMaior || _EhMenor,
                                    "600",
                                    "400"
                                )
                                &
                            "'
                            fill='" & _CorDestaque & "'
                        >
                            " & FORMAT ( _Valor, "0.#" ) & "%
                        </text>
                        "
                    )

                    &

                    IF (
                        ( _EhMaior || _EhMenor )
                            && _QtdIndicadores = 1,

                        "
                        <text
                            x='" & FORMAT ( _X, "0.0", "en-US" ) & "'
                            y='" & FORMAT ( _Y - 27, "0.0", "en-US" ) & "'
                            text-anchor='middle'
                            font-family='Jost,Segoe UI,sans-serif'
                            font-size='6.5'
                            font-weight='500'
                            letter-spacing='1.2'
                            fill='" & _CorDestaque & "'
                            opacity='.7'
                        >
                            " & _LabelEspecial & "
                        </text>
                        "
