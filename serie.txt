VAR _IndicadoresSelecionados =
    VALUES ( 'Parâmetro'[Parâmetro] )

/* =========================================================
   PERÍODOS VISÍVEIS
   Mantém os filtros da página, mas recupera os meses do
   período selecionado.
   ========================================================= */
VAR _Periodos =
    CALCULATETABLE (
        SUMMARIZE (
            'BASE',
            'BASE'[ANO],
            'BASE'[MESANO]
        ),
        ALLSELECTED ( 'BASE'[ANO], 'BASE'[MESANO] )
    )

/* =========================================================
   LABELS DO EIXO X
   ========================================================= */
VAR _Labels =
    CONCATENATEX (
        _Periodos,
        "'" &
            SUBSTITUTE (
                FORMAT ( 'BASE'[MESANO], "mmm/yyyy" ),
                "'",
                ""
            ) &
        "'",
        ",",
        'BASE'[MESANO],
        ASC
    )

VAR _Years =
    CONCATENATEX (
        _Periodos,
        FORMAT ( 'BASE'[ANO], "0" ),
        ",",
        'BASE'[MESANO],
        ASC
    )

/* =========================================================
   MONTA CADA SÉRIE DO FIELD PARAMETER
   ========================================================= */
VAR _Series =
    CONCATENATEX (
        _IndicadoresSelecionados,

        VAR _Indicador =
            'Parâmetro'[Parâmetro]

        VAR _Ordem =
            SWITCH (
                _Indicador,
                "Resultado - ROB Crédito", 0,
                "Resultado - ROB Não crédito", 1,
                "Resultado - TPV", 2,
                "Resultado - Fat. Líquido", 3,
                "Consistência - Encanta/NPS", 4,
                "Consistência - BSC", 5,
                "PDD", 6,
                99
            )

        VAR _Cor =
            SWITCH (
                _Indicador,
                "Resultado - ROB Crédito", "#0066B3",
                "Resultado - ROB Não crédito", "#17365D",
                "Resultado - TPV", "#43A6C6",
                "Resultado - Fat. Líquido", "#89BCD3",
                "Consistência - Encanta/NPS", "#276749",
                "Consistência - BSC", "#548235",
                "PDD", "#767171",
                "#666666"
            )

        VAR _DadosIndicador =
            ADDCOLUMNS (
                _Periodos,
                "@Valor",
                    SWITCH (
                        _Indicador,

                        "Resultado - ROB Crédito",
                            CALCULATE ( 'BASE'[% Atingimento ROB Crédito] ),

                        "Resultado - ROB Não crédito",
                            CALCULATE ( 'BASE'[% Atingimento ROB Não Crédito] ),

                        "Resultado - TPV",
                            CALCULATE ( 'BASE'[% Atingimento TPV] ),

                        "Resultado - Fat. Líquido",
                            CALCULATE ( 'BASE'[% Atingimento Faturamento Liquido] ),

                        "Consistência - Encanta/NPS",
                            CALCULATE ( 'BASE'[% Atingimento Encanta BRA] ),

                        "Consistência - BSC",
                            CALCULATE ( 'BASE'[% Atingimento BSC] ),

                        "PDD",
                            CALCULATE ( 'BASE'[% Atingimento PDD] )
                    )
            )

        VAR _Valores =
            CONCATENATEX (
                _DadosIndicador,
                VAR _V = [@Valor]
                RETURN
                    IF (
                        ISBLANK ( _V ),
                        "null",
                        FORMAT ( _V * 100, "0.########", "en-US" )
                    ),
                ",",
                'BASE'[MESANO],
                ASC
            )

        RETURN
            "{" &
                "name:'" & SUBSTITUTE ( _Indicador, "'", "" ) & "'," &
                "order:" & FORMAT ( _Ordem, "0" ) & "," &
                "color:'" & _Cor & "'," &
                "values:[" & _Valores & "]" &
            "}",

        ",",
        SWITCH (
            'Parâmetro'[Parâmetro],
            "Resultado - ROB Crédito", 0,
            "Resultado - ROB Não crédito", 1,
            "Resultado - TPV", 2,
            "Resultado - Fat. Líquido", 3,
            "Consistência - Encanta/NPS", 4,
            "Consistência - BSC", 5,
            "PDD", 6,
            99
        ),
        ASC
    )

VAR _QtdIndicadores =
    COUNTROWS ( _IndicadoresSelecionados )

VAR _Titulo =
    "Percentuais de Atingimentos"

VAR _Subtitulo =
    "Atingimentos dos indicadores por mês e ano"

RETURN

    stroke:#E8E8E8;
    stroke-width:1;

}

    position:absolute;
    display:none;
    pointer-events:none;
    min-width:135px;
    background:rgba(255,255,255,.97);
    border:1px solid #E0E0E0;
    border-radius:7px;
    padding:8px 10px;
    box-shadow:0 3px 12px rgba(0,0,0,.12);
    z-index:20;
}

    font-size:10px;
    color:#777;
    margin-bottom:5px;
}

.tip-name{
    font-size:10px;

}

    font-size:14px;

}

<body>

<div class='card'>

    <div class='header'>

        <div class='title'>" & _Titulo & "</div>

        <div class='subtitle'>" & _Subtitulo & "</div>

        <div id='legend' class='legend'></div>

    </div>

    <div id='chart' class='chart-container'>
        <div id='tooltip'></div>
    </div>

</div>

<script>

(function(){

const labels = [" & _Labels & "];

const years = [" & _Years & "];

const series = [" & _Series & "];

if(labels.length === 0 || series.length === 0)
    return;


/* =========================================================
   LEGENDA
   ========================================================= */

const legend = document.getElementById('legend');

series.forEach(s => {

    const item = document.createElement('div');
    item.className = 'legend-item';

    item.innerHTML =
        `<span class='legend-dot'
        style='background:${s.color}'></span>
        <span>${s.name}</span>`;

    legend.appendChild(item);

});


/* =========================================================
   TODOS OS VALORES
   ========================================================= */

let allValues=[];

series.forEach(s => {
    s.values.forEach(v => {
        if(v !== null && !isNaN(v))
            allValues.push(v);
    });
});

if(allValues.length === 0)
    return;


let minVal = Math.min(...allValues);
let maxVal = Math.max(...allValues);


/* Sempre mostrar pelo menos a região de 100% */

minVal = Math.min(minVal,95);
maxVal = Math.max(maxVal,105);

let range = maxVal-minVal;

if(range <= 0)
    range=10;

let padding = range*0.13;

let yMin = minVal-padding;
let yMax = maxVal+padding;


/* =========================================================
   TAMANHO
   ========================================================= */

const container =
    document.getElementById('chart');

const W =
    container.clientWidth || 1200;

const H =
    container.clientHeight || 400;

const pad={
    left:52,
    right:28,
    top:32,
    bottom:55
};

const chartW =
    W-pad.left-pad.right;

const chartH =
    H-pad.top-pad.bottom;


/* =========================================================
   SVG
   ========================================================= */

const NS =
    'http://www.w3.org/2000/svg';

const svg =
    document.createElementNS(NS,'svg');

svg.setAttribute(
    'viewBox',
    `0 0 ${W} ${H}`
);

container.appendChild(svg);


/* =========================================================
   FUNÇÕES
   ========================================================= */

function xPos(i){

    if(labels.length===1)
        return pad.left+chartW/2;

    return pad.left+
        (i/(labels.length-1))*chartW;
}

function yPos(v){

    return pad.top+
        chartH-
        ((v-yMin)/(yMax-yMin))*chartH;
}


/* Curva suave */

function smoothPath(points){

    if(points.length===0)
        return '';

    if(points.length===1)
        return `M${points[0].x},${points[0].y}`;

    let d =
        `M${points[0].x},${points[0].y}`;

    for(let i=0;i<points.length-1;i++){

        const p0 =
            points[Math.max(0,i-1)];

        const p1 =
            points[i];

        const p2 =
            points[i+1];

        const p3 =
            points[Math.min(points.length-1,i+2)];

        const tension=.20;

        const cp1x =
            p1.x+(p2.x-p0.x)*tension;

        const cp1y =
            p1.y+(p2.y-p0.y)*tension;

        const cp2x =
            p2.x-(p3.x-p1.x)*tension;

        const cp2y =
            p2.y-(p3.y-p1.y)*tension;

        d +=
            ` C${cp1x},${cp1y}
               ${cp2x},${cp2y}
               ${p2.x},${p2.y}`;
    }

    return d;
}


/* =========================================================
   GRID
   ========================================================= */

const gridCount=5;

for(let i=0;i<gridCount;i++){

    const value =
        yMin+
        ((yMax-yMin)/(gridCount-1))*i;

    const y =
        yPos(value);

    const line =
        document.createElementNS(NS,'line');

    line.setAttribute('x1',pad.left);
    line.setAttribute('x2',W-pad.right);
    line.setAttribute('y1',y);
    line.setAttribute('y2',y);
    line.setAttribute('class','grid');

    svg.appendChild(line);

}


/* =========================================================
   LINHA DE 100%
   ========================================================= */

if(100>=yMin && 100<=yMax){

    const y100=yPos(100);

    const line =
        document.createElementNS(NS,'line');

    line.setAttribute('x1',pad.left);
    line.setAttribute('x2',W-pad.right);
    line.setAttribute('y1',y100);
    line.setAttribute('y2',y100);
    line.setAttribute('class','target');

    svg.appendChild(line);


    const txt =
        document.createElementNS(NS,'text');

    txt.setAttribute('x',pad.left-8);
    txt.setAttribute('y',y100+4);
    txt.setAttribute('text-anchor','end');

    txt.setAttribute(
        'style',
        'font-size:10px;font-weight:600;fill:#A33C36'
    );

    txt.textContent='100%';

    svg.appendChild(txt);
}


/* =========================================================
   EIXO X
   ========================================================= */

labels.forEach((label,i)=>{

    const x=xPos(i);

    const txt =
        document.createElementNS(NS,'text');

    txt.setAttribute('x',x);
    txt.setAttribute(
        'y',
        H-pad.bottom+20
    );

    txt.setAttribute(
        'text-anchor',
        'middle'
    );

    txt.setAttribute(
        'class',
        'axis-label'
    );

    txt.textContent=label;

    svg.appendChild(txt);

});


/* =========================================================
   AGRUPAMENTO DOS ANOS
   ========================================================= */

let yearGroups=[];

years.forEach((year,i)=>{

    const last =
        yearGroups[
            yearGroups.length-1
        ];

    if(!last || last.year!==year){

        yearGroups.push({
            year:year,
            start:i,
            end:i
        });

    }else{

        last.end=i;

    }

});


yearGroups.forEach(g=>{

    const x1=xPos(g.start);
    const x2=xPos(g.end);

    const mid=(x1+x2)/2;

    const txt =
        document.createElementNS(
            NS,
            'text'
        );

    txt.setAttribute('x',mid);

    txt.setAttribute(
        'y',
        H-7
    );

    txt.setAttribute(
        'text-anchor',
        'middle'
    );

    txt.setAttribute(
        'class',
        'year-label'
    );

    txt.textContent=g.year;

    svg.appendChild(txt);

});


/* =========================================================
   TOOLTIP
   ========================================================= */

const tooltip =
    document.getElementById('tooltip');


/* =========================================================
   SÉRIES
   ========================================================= */

series.forEach((s,seriesIndex)=>{

    let pts=[];

    s.values.forEach((v,i)=>{

        if(v!==null){

            pts.push({
                x:xPos(i),
                y:yPos(v),
                value:v,
                index:i
            });

        }

    });


    if(pts.length===0)
        return;


    /* linha */

    const path =
        document.createElementNS(
            NS,
            'path'
        );

    path.setAttribute(
        'd',
        smoothPath(pts)
    );

    path.setAttribute(
        'stroke',
        s.color
    );

    path.setAttribute(
        'class',
        'series-line'
    );

    svg.appendChild(path);


    /* pontos */

    pts.forEach(p=>{

        const circle =
            document.createElementNS(
                NS,
                'circle'
            );

        circle.setAttribute(
            'cx',
            p.x
        );

        circle.setAttribute(
            'cy',
            p.y
        );

        circle.setAttribute(
            'r',
            3.4
        );

        circle.setAttribute(
            'fill',
            s.color
        );

        circle.setAttribute(
            'class',
            'point'
        );

        svg.appendChild(circle);


        /* label */

        const label =
            document.createElementNS(
                NS,
                'text'
            );

        label.setAttribute(
            'x',
            p.x
        );

        /*
          distribui labels entre as séries
          para reduzir sobreposição
        */

        const offset =
            11 +
            (seriesIndex%2)*11;

        label.setAttribute(
            'y',
            p.y-offset
        );

        label.setAttribute(
            'text-anchor',
            'middle'
        );

        label.setAttribute(
            'fill',
            s.color
        );

        label.setAttribute(
            'class',
            'data-label'
        );

        label.textContent =
            p.value.toFixed(
                p.value%1===0 ? 0 : 1
            ) + '%';

        svg.appendChild(label);


        /* área maior para mouse */

        const hit =
            document.createElementNS(
                NS,
                'circle'
            );

        hit.setAttribute('cx',p.x);
        hit.setAttribute('cy',p.y);
        hit.setAttribute('r',12);
        hit.setAttribute(
            'fill',
            'transparent'
        );

        hit.style.cursor='pointer';


        hit.addEventListener(
            'mouseenter',
            function(){

                tooltip.innerHTML =

                    `<div class='tip-period'>
                        ${labels[p.index]}
                    </div>

                    <div class='tip-name'>
                        ${s.name}
                    </div>

                    <div class='tip-value'
                         style='color:${s.color}'>
                        ${p.value.toFixed(1)}%
                    </div>`;

                tooltip.style.display =
                    'block';

                let left=p.x+12;
                let top=p.y-35;

                if(left+160>W)
                    left=p.x-160;

                tooltip.style.left =
                    left+'px';

                tooltip.style.top =
                    top+'px';

            }
        );


        hit.addEventListener(
            'mouseleave',
            function(){

                tooltip.style.display =
                    'none';

            }
        );


        svg.appendChild(hit);

    });

});

})();
</script>

</body>
</html>"
