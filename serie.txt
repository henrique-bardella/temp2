Histórico Indicadores HTML =
VAR _W = 1600
VAR _SvgH = 430

VAR _PadLeft = 48
VAR _PadRight = 30
VAR _PadTop = 40
VAR _PadBottom = 72

VAR _ChartW = _W - _PadLeft - _PadRight
VAR _ChartH = _SvgH - _PadTop - _PadBottom

/* ============================================================
   CORES
   ============================================================ */
VAR _CorROB_Credito       = "#1E63B5"
VAR _CorROB_NaoCredito    = "#1B2F8A"
VAR _CorTPV               = "#8ED8F8"
VAR _CorFatLiquido        = "#A8B9EE"
VAR _CorBSC               = "#57B34C"
VAR _CorEncanta           = "#255C2B"
VAR _CorPDD               = "#7F7F7F"

VAR _CorTextoPrincipal    = "#222222"
VAR _CorTextoSecundario   = "#666666"
VAR _CorGrid              = "#E5E5E5"
VAR _CorMeta              = "#B04343"
VAR _CorDataFuncao        = "#C96A3A"

/* ============================================================
   FIELD PARAMETER
   ============================================================ */
VAR _IndicadoresSelecionados =
    SELECTCOLUMNS (
        SUMMARIZE (
            'Parâmetro',
            'Parâmetro'[Parâmetro],
            'Parâmetro'[Parâmetro Campos],
            'Parâmetro'[Parâmetro Pedido]
        ),
        "@Indicador", 'Parâmetro'[Parâmetro],
        "@Ordem", 'Parâmetro'[Parâmetro Pedido]
    )

VAR _QtdIndicadores =
    COUNTROWS ( _IndicadoresSelecionados )

/* ============================================================
   PERÍODOS
   ============================================================ */
VAR _PeriodosBase =
    CALCULATETABLE (
        SUMMARIZE (
            'BASE',
            'BASE'[ANO],
            'BASE'[MESANO]
        ),
        ALLSELECTED (
            'BASE'[ANO],
            'BASE'[MESANO]
        )
    )

/* ============================================================
   TRATA MÊS / ORDEM CRONOLÓGICA
   Assumindo MESANO textual como jan/2025, fev/2025...
   ============================================================ */
VAR _PeriodosComMes =
    ADDCOLUMNS (
        _PeriodosBase,
        "@MesTexto",
            LOWER ( TRIM ( 'BASE'[MESANO] & "" ) ),
        "@MesNum",
            VAR _M = LOWER ( TRIM ( 'BASE'[MESANO] & "" ) )
            RETURN
                SWITCH (
                    TRUE (),
                    LEFT ( _M, 3 ) = "jan", 1,
                    LEFT ( _M, 3 ) = "fev", 2,
                    LEFT ( _M, 3 ) = "mar", 3,
                    LEFT ( _M, 3 ) = "abr", 4,
                    LEFT ( _M, 3 ) = "mai", 5,
                    LEFT ( _M, 3 ) = "jun", 6,
                    LEFT ( _M, 3 ) = "jul", 7,
                    LEFT ( _M, 3 ) = "ago", 8,
                    LEFT ( _M, 3 ) = "set", 9,
                    LEFT ( _M, 3 ) = "out", 10,
                    LEFT ( _M, 3 ) = "nov", 11,
                    LEFT ( _M, 3 ) = "dez", 12,
                    BLANK ()
                )
    )

VAR _PeriodosSort =
    ADDCOLUMNS (
        _PeriodosComMes,
        "@Sort", VALUE ( 'BASE'[ANO] ) * 100 + [@MesNum],
        "@DataRef", DATE ( VALUE ( 'BASE'[ANO] ), [@MesNum], 1 )
    )

VAR _Periodos =
    ADDCOLUMNS (
        _PeriodosSort,
        "@Pos",
            VAR _SortAtual = [@Sort]
            RETURN
                RANKX (
                    _PeriodosSort,
                    [@Sort],
                    _SortAtual,
                    ASC,
                    DENSE
                )
    )

VAR _QtdPeriodos =
    COUNTROWS ( _Periodos )

VAR _DivX =
    MAX ( 1, _QtdPeriodos - 1 )

/* ============================================================
   DATA INÍCIO DA FUNÇÃO
   ============================================================ */
VAR _DataFuncao =
    SELECTEDVALUE ( 'dimGerentes2'[data_funcao] )

VAR _DataFuncaoMes =
    IF (
        NOT ISBLANK ( _DataFuncao ),
        DATE ( YEAR ( _DataFuncao ), MONTH ( _DataFuncao ), 1 )
    )

VAR _MinSerieData =
    MINX ( _Periodos, [@DataRef] )

VAR _MaxSerieData =
    MAXX ( _Periodos, [@DataRef] )

VAR _TemDataFuncaoNaSerie =
    NOT ISBLANK ( _DataFuncaoMes )
        && _DataFuncaoMes >= _MinSerieData
        && _DataFuncaoMes <= _MaxSerieData

VAR _PosDataFuncao =
    MINX (
        TOPN (
            1,
            FILTER ( _Periodos, [@DataRef] >= _DataFuncaoMes ),
            [@DataRef], ASC
        ),
        [@Pos]
    )

/* ============================================================
   TABELA LONGA
   ============================================================ */
VAR _DadosLongos =
    GENERATE (
        _IndicadoresSelecionados,

        VAR _NomeIndicador = [@Indicador]
        VAR _OrdemIndicador = [@Ordem]

        VAR _CorIndicador =
            SWITCH (
                _NomeIndicador,
                "Resultado - ROB Crédito", _CorROB_Credito,
                "Resultado - ROB Não crédito", _CorROB_NaoCredito,
                "Resultado - TPV", _CorTPV,
                "Resultado - Fat. Líquido", _CorFatLiquido,
                "Consistência - BSC", _CorBSC,
                "Consistência - Encanta/NPS", _CorEncanta,
                "PDD", _CorPDD,
                "#777777"
            )

        RETURN
            SELECTCOLUMNS (
                ADDCOLUMNS (
                    _Periodos,
                    "@ValorTemp",
                        SWITCH (
                            _NomeIndicador,
                            "Resultado - ROB Crédito",
                                CALCULATE ( 'BASE'[% Atingimento ROB Crédito] ) * 100,
                            "Resultado - ROB Não crédito",
                                CALCULATE ( 'BASE'[% Atingimento ROB Não Crédito] ) * 100,
                            "Resultado - TPV",
                                CALCULATE ( 'BASE'[% Atingimento TPV] ) * 100,
                            "Resultado - Fat. Líquido",
                                CALCULATE ( 'BASE'[% Atingimento Faturamento Líquido] ) * 100,
                            "Consistência - BSC",
                                CALCULATE ( 'BASE'[% Atingimento BSC] ) * 100,
                            "Consistência - Encanta/NPS",
                                CALCULATE ( 'BASE'[% Atingimento Encanta BRA] ) * 100,
                            "PDD",
                                CALCULATE ( 'BASE'[% Atingimento PDD] ) * 100
                        )
                ),
                "@IndicadorL", _NomeIndicador,
                "@OrdemL", _OrdemIndicador,
                "@CorL", _CorIndicador,
                "@PosL", [@Pos],
                "@SortL", [@Sort],
                "@AnoL", 'BASE'[ANO],
                "@MesNumL", [@MesNum],
                "@MesAnoL", 'BASE'[MESANO],
                "@DataRefL", [@DataRef],
                "@ValorL", [@ValorTemp]
            )
    )

VAR _DadosValidos =
    FILTER ( _DadosLongos, NOT ISBLANK ( [@ValorL] ) )

VAR _TemDados =
    COUNTROWS ( _DadosValidos ) > 0

/* ============================================================
   ESCALA Y
   ============================================================ */
VAR _MinData =
    MINX ( _DadosValidos, [@ValorL] )

VAR _MaxData =
    MAXX ( _DadosValidos, [@ValorL] )

VAR _MinBase =
    MIN ( _MinData, 100 )

VAR _MaxBase =
    MAX ( _MaxData, 100 )

VAR _Amplitude =
    MAX ( 10, _MaxBase - _MinBase )

VAR _YMin =
    MAX ( 0, _MinBase - _Amplitude * 0.18 )

VAR _YMax =
    _MaxBase + _Amplitude * 0.22

VAR _YRange =
    MAX ( 1, _YMax - _YMin )

/* ============================================================
   GRID
   ============================================================ */
VAR _Grid =
    CONCATENATEX (
        GENERATESERIES ( 0, 3, 1 ),
        VAR _I = [Value]
        VAR _V = _YMin + DIVIDE ( _YRange, 3 ) * _I
        VAR _Y =
            _PadTop
                + _ChartH
                - DIVIDE ( _V - _YMin, _YRange ) * _ChartH
        RETURN
            "<line x1='" & FORMAT ( _PadLeft, "0.0", "en-US" ) &
            "' x2='" & FORMAT ( _W - _PadRight, "0.0", "en-US" ) &
            "' y1='" & FORMAT ( _Y, "0.0", "en-US" ) &
            "' y2='" & FORMAT ( _Y, "0.0", "en-US" ) &
            "' stroke='" & _CorGrid & "' stroke-width='1' stroke-dasharray='2 4' />",
        ""
    )

/* ============================================================
   LINHA META 100%
   ============================================================ */
VAR _Y100 =
    _PadTop
        + _ChartH
        - DIVIDE ( 100 - _YMin, _YRange ) * _ChartH

VAR _LinhaMeta =
    "<line x1='" & FORMAT ( _PadLeft, "0", "en-US" ) &
    "' x2='" & FORMAT ( _W - _PadRight, "0", "en-US" ) &
    "' y1='" & FORMAT ( _Y100, "0.0", "en-US" ) &
    "' y2='" & FORMAT ( _Y100, "0.0", "en-US" ) &
    "' stroke='" & _CorMeta & "' stroke-width='2' stroke-dasharray='6 5' opacity='0.9' />"

/* ============================================================
   MARCADOR DATA FUNÇÃO
   ============================================================ */
VAR _MarcadorDataFuncao =
    IF (
        _TemDataFuncaoNaSerie && NOT ISBLANK ( _PosDataFuncao ),
        VAR _XData =
            _PadLeft
                + DIVIDE ( _PosDataFuncao - 1, _DivX ) * _ChartW
        VAR _Label1 = "Início da função"
        VAR _Label2 = FORMAT ( _DataFuncao, "dd/MM/yyyy" )
        VAR _BoxW =
            MAX ( 150, LEN ( _Label2 ) * 8 + 34 )
        VAR _BoxX = _XData - DIVIDE ( _BoxW, 2 )
        RETURN
            "<line x1='" & FORMAT ( _XData, "0.0", "en-US" ) &
            "' x2='" & FORMAT ( _XData, "0.0", "en-US" ) &
            "' y1='" & FORMAT ( _PadTop, "0.0", "en-US" ) &
            "' y2='" & FORMAT ( _PadTop + _ChartH, "0.0", "en-US" ) &
            "' stroke='" & _CorDataFuncao & "' stroke-width='2' stroke-dasharray='5 4' opacity='0.95' />" &

            "<circle cx='" & FORMAT ( _XData, "0.0", "en-US" ) &
            "' cy='" & FORMAT ( _PadTop, "0.0", "en-US" ) &
            "' r='5' fill='" & _CorDataFuncao & "' />" &

            "<rect x='" & FORMAT ( _BoxX, "0.0", "en-US" ) &
            "' y='4' width='" & FORMAT ( _BoxW, "0.0", "en-US" ) &
            "' height='34' rx='8' fill='white' stroke='" & _CorDataFuncao &
            "' stroke-width='1.5' opacity='0.96' />" &

            "<text x='" & FORMAT ( _XData, "0.0", "en-US" ) &
            "' y='17' text-anchor='middle' font-family='Segoe UI' font-size='11' font-weight='700' fill='" & _CorDataFuncao & "'>" &
            _Label1 &
            "</text>" &

            "<text x='" & FORMAT ( _XData, "0.0", "en-US" ) &
            "' y='30' text-anchor='middle' font-family='Segoe UI' font-size='11' font-weight='600' fill='" & _CorDataFuncao & "'>" &
            _Label2 &
            "</text>"
    )

/* ============================================================
   EIXO X
   ============================================================ */
VAR _MesesSVG =
    CONCATENATEX (
        _Periodos,
        VAR _Pos = [@Pos]
        VAR _X =
            _PadLeft
                + DIVIDE ( _Pos - 1, _DivX ) * _ChartW
        VAR _MesLabel =
            SWITCH (
                [@MesNum],
                1, "jan",
                2, "fev",
                3, "mar",
                4, "abr",
                5, "mai",
                6, "jun",
                7, "jul",
                8, "ago",
                9, "set",
                10, "out",
                11, "nov",
                12, "dez"
            ) & "/" & FORMAT ( 'BASE'[ANO], "0" )
        RETURN
            "<text x='" & FORMAT ( _X, "0.0", "en-US" ) &
            "' y='" & FORMAT ( _SvgH - 28, "0", "en-US" ) &
            "' text-anchor='middle' font-family='Segoe UI' font-size='12' font-weight='500' fill='" & _CorTextoSecundario & "'>" &
            _MesLabel &
            "</text>",
        "",
        [@Sort],
        ASC
    )

VAR _Anos =
    DISTINCT (
        SELECTCOLUMNS (
            _Periodos,
            "@Ano", 'BASE'[ANO]
        )
    )

VAR _AnosSVG =
    CONCATENATEX (
        _Anos,
        VAR _AnoAtual = [@Ano]
        VAR _Inicio =
            MINX ( FILTER ( _Periodos, 'BASE'[ANO] = _AnoAtual ), [@Pos] )
        VAR _Fim =
            MAXX ( FILTER ( _Periodos, 'BASE'[ANO] = _AnoAtual ), [@Pos] )
        VAR _X1 = _PadLeft + DIVIDE ( _Inicio - 1, _DivX ) * _ChartW
        VAR _X2 = _PadLeft + DIVIDE ( _Fim - 1, _DivX ) * _ChartW
        VAR _XC = DIVIDE ( _X1 + _X2, 2 )
        RETURN
            "<text x='" & FORMAT ( _XC, "0.0", "en-US" ) &
            "' y='" & FORMAT ( _SvgH - 6, "0", "en-US" ) &
            "' text-anchor='middle' font-family='Segoe UI' font-size='13' font-weight='700' fill='" & _CorTextoPrincipal & "'>" &
            FORMAT ( _AnoAtual, "0" ) &
            "</text>",
        ""
    )

/* ============================================================
   LEGENDA
   ============================================================ */
VAR _Legenda =
    CONCATENATEX (
        _IndicadoresSelecionados,
        VAR _Nome = [@Indicador]
        VAR _Cor =
            SWITCH (
                _Nome,
                "Resultado - ROB Crédito", _CorROB_Credito,
                "Resultado - ROB Não crédito", _CorROB_NaoCredito,
                "Resultado - TPV", _CorTPV,
                "Resultado - Fat. Líquido", _CorFatLiquido,
                "Consistência - BSC", _CorBSC,
                "Consistência - Encanta/NPS", _CorEncanta,
                "PDD", _CorPDD,
                "#777777"
            )
        RETURN
            "<div class='legend-item'>" &
                "<span class='legend-dot' style='background:" & _Cor & ";'></span>" &
                "<span>" & _Nome & "</span>" &
            "</div>",
        "",
        [@Ordem],
        ASC
    )

/* ============================================================
   SÉRIES
   ============================================================ */
VAR _SeriesSVG =
    CONCATENATEX (
        _IndicadoresSelecionados,

        VAR _Nome = [@Indicador]
        VAR _Ordem = [@Ordem]

        VAR _Cor =
            SWITCH (
                _Nome,
                "Resultado - ROB Crédito", _CorROB_Credito,
                "Resultado - ROB Não crédito", _CorROB_NaoCredito,
                "Resultado - TPV", _CorTPV,
                "Resultado - Fat. Líquido", _CorFatLiquido,
                "Consistência - BSC", _CorBSC,
                "Consistência - Encanta/NPS", _CorEncanta,
                "PDD", _CorPDD,
                "#777777"
            )

        VAR _OffsetLabel =
            SWITCH (
                MOD ( _Ordem, 7 ),
                0, -18,
                1, 18,
                2, -34,
                3, 34,
                4, -50,
                5, 50,
                6, -66,
                -18
            )

        VAR _Serie =
            FILTER ( _DadosValidos, [@IndicadorL] = _Nome )

        VAR _Points =
            CONCATENATEX (
                _Serie,
                VAR _X =
                    _PadLeft
                        + DIVIDE ( [@PosL] - 1, _DivX ) * _ChartW
                VAR _Y =
                    _PadTop
                        + _ChartH
                        - DIVIDE ( [@ValorL] - _YMin, _YRange ) * _ChartH
                RETURN
                    FORMAT ( _X, "0.0", "en-US" ) &
                    "," &
                    FORMAT ( _Y, "0.0", "en-US" ),
                " ",
                [@SortL],
                ASC
            )

        VAR _Detalhes =
            CONCATENATEX (
                _Serie,
                VAR _Valor = [@ValorL]
                VAR _TextoValor = FORMAT ( _Valor, "0.#" ) & "%"
                VAR _X =
                    _PadLeft
                        + DIVIDE ( [@PosL] - 1, _DivX ) * _ChartW
                VAR _Y =
                    _PadTop
                        + _ChartH
                        - DIVIDE ( _Valor - _YMin, _YRange ) * _ChartH

                VAR _LabelY = _Y + _OffsetLabel
                VAR _BoxW = LEN ( _TextoValor ) * 9 + 16
                VAR _BoxH = 22
                VAR _BoxX = _X - DIVIDE ( _BoxW, 2 )
                VAR _BoxY = _LabelY - 16

                RETURN
                    "<circle cx='" & FORMAT ( _X, "0.0", "en-US" ) &
                    "' cy='" & FORMAT ( _Y, "0.0", "en-US" ) &
                    "' r='4.2' fill='" & _Cor & "' stroke='white' stroke-width='1.6' />" &

                    "<rect x='" & FORMAT ( _BoxX, "0.0", "en-US" ) &
                    "' y='" & FORMAT ( _BoxY, "0.0", "en-US" ) &
                    "' width='" & FORMAT ( _BoxW, "0.0", "en-US" ) &
                    "' height='" & FORMAT ( _BoxH, "0.0", "en-US" ) &
                    "' rx='5' fill='white' opacity='0.88' />" &

                    "<text x='" & FORMAT ( _X, "0.0", "en-US" ) &
                    "' y='" & FORMAT ( _LabelY, "0.0", "en-US" ) &
                    "' text-anchor='middle' font-family='Segoe UI' font-size='14' font-weight='700' fill='" & _Cor & "'>" &
                    _TextoValor &
                    "</text>",
                "",
                [@SortL],
                ASC
            )

        RETURN
            "<polyline points='" & _Points &
            "' fill='none' stroke='" & _Cor &
            "' stroke-width='2.4' stroke-linecap='round' stroke-linejoin='round' stroke-dasharray='7 5' opacity='0.96' />"
            & _Detalhes,

        "",
        [@Ordem],
        ASC
    )

/* ============================================================
   HTML FINAL
   ============================================================ */
RETURN
IF (
    NOT _TemDados,
    "
    <div style='
        width:100%;
        height:640px;
        display:flex;
        align-items:center;
        justify-content:center;
        font-family:Segoe UI;
        color:#777;
        font-size:14px;
    '>
        Não há dados para os filtros selecionados.
    </div>
    ",
    "
    <div class='tc'>
        <style>
            .tc{
                width:100%;
                height:640px;
                background:#FFFFFF;
                font-family:'Segoe UI',Arial,sans-serif;
                color:" & _CorTextoPrincipal & ";
                padding:18px 20px 14px 20px;
                display:flex;
                flex-direction:column;
                box-sizing:border-box;
                overflow:hidden;
            }

            .header{
                flex-shrink:0;
            }

            .section-title{
                font-size:16px;
                font-weight:700;
                line-height:1.2;
                margin-bottom:6px;
                color:" & _CorTextoPrincipal & ";
            }

            .title{
                font-size:16px;
                font-weight:800;
                line-height:1.2;
                color:" & _CorTextoPrincipal & ";
            }

            .subtitle{
                font-size:14px;
                font-weight:500;
                line-height:1.3;
                margin-top:4px;
                color:" & _CorTextoSecundario & ";
            }

            .legend{
                display:flex;
                flex-wrap:wrap;
                gap:10px 18px;
                align-items:center;
                margin-top:12px;
                margin-bottom:8px;
            }

            .legend-item{
                display:flex;
                align-items:center;
                gap:6px;
                font-size:14px;
                font-weight:500;
                color:" & _CorTextoPrincipal & ";
                white-space:nowrap;
            }

            .legend-dot{
                width:10px;
                height:10px;
                min-width:10px;
                border-radius:50%;
                display:inline-block;
            }

            .chart-wrap{
                flex:1;
                min-height:0;
            }

            svg{
                width:100%;
                height:100%;
                display:block;
            }
        </style>

        <div class='header'>
            <div class='section-title'>Histórico dos indicadores</div>
            <div class='title'>Percentuais de Atingimentos</div>
            <div class='subtitle'>Atingimentos dos indicadores por mês e Ano</div>
            <div class='legend'>
                " & _Legenda & "
            </div>
        </div>

        <div class='chart-wrap'>
            <svg viewBox='0 0 " & FORMAT ( _W, "0", "en-US" ) & " " & FORMAT ( _SvgH, "0", "en-US" ) & "' preserveAspectRatio='none'>
                " &
                _Grid &
                _LinhaMeta &
                _MarcadorDataFuncao &
                _SeriesSVG &
                _MesesSVG &
                _AnosSVG &
                "
            </svg>
        </div>
    </div>
    "
)
